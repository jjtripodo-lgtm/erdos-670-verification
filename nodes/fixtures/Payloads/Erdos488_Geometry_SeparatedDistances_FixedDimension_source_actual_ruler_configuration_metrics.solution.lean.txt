/-- A two-dimensional finite-difference sum telescopes to its four corners. -/
theorem rectangular_mixed_difference_telescope (F : ℕ → ℕ → ℝ)
    {i j : ℕ} (hij : i ≤ j) :
    (∑ a ∈ Finset.Ico i j, ∑ b ∈ Finset.Ico i j,
      (F (a + 1) (b + 1) - F (a + 1) b - F a (b + 1) + F a b)) =
      F j j - F j i - F i j + F i i := by
  have hinner (a : ℕ) :
      (∑ b ∈ Finset.Ico i j,
        (F (a + 1) (b + 1) - F (a + 1) b - F a (b + 1) + F a b)) =
        (F (a + 1) j - F (a + 1) i) - (F a j - F a i) := by
    have h1 := Finset.sum_Ico_sub (fun b => F (a + 1) b) hij
    have h2 := Finset.sum_Ico_sub (fun b => F a b) hij
    rw [← h1, ← h2, ← Finset.sum_sub_distrib]
    apply Finset.sum_congr rfl
    intro b hb
    ring
  simp_rw [hinner]
  have hout := Finset.sum_Ico_sub (fun a => F a j - F a i) hij
  rw [hout]
  ring

/-- Symmetric squared-distance data produce a symmetric increment Gram matrix. -/
theorem incrementGram_isHermitian {n : ℕ} (D : SqDistanceData (Fin (n + 1)))
    (hD : D.IsHermitian) : (incrementGram D).IsHermitian := by
  apply Matrix.IsHermitian.ext
  intro i j
  simp only [incrementGram, starRingEnd_apply, star_trivial]
  have h11 := hD.apply i.succ j.succ
  have h10 := hD.apply i.succ j.castSucc
  have h01 := hD.apply i.castSucc j.succ
  have h00 := hD.apply i.castSucc j.castSucc
  simp only [star_trivial] at h11 h10 h01 h00
  rw [← h11, ← h10, ← h01, ← h00]
  ring

/-- For symmetric zero-diagonal data, the block mixed difference is the endpoint value. -/
theorem mixed_difference_block_eq_distance (D : ℕ → ℕ → ℝ)
    (hD : ∀ i j, D i j = D j i) (hdiag : ∀ i, D i i = 0)
    {i j : ℕ} (hij : i ≤ j) :
    (∑ a ∈ Finset.Ico i j, ∑ b ∈ Finset.Ico i j,
      (-(1 / 2 : ℝ) *
        (D (a + 1) (b + 1) - D (a + 1) b - D a (b + 1) + D a b))) =
      D i j := by
  calc
    _ = -(1 / 2 : ℝ) *
        (∑ a ∈ Finset.Ico i j, ∑ b ∈ Finset.Ico i j,
          (D (a + 1) (b + 1) - D (a + 1) b - D a (b + 1) + D a b)) := by
          rw [Finset.mul_sum]
          apply Finset.sum_congr rfl
          intro a ha
          rw [Finset.mul_sum]
    _ = -(1 / 2 : ℝ) * (D j j - D j i - D i j + D i i) := by
          rw [rectangular_mixed_difference_telescope D hij]
    _ = D i j := by
      rw [hdiag, hdiag, hD j i]
      ring



/-- A positive-definite Gram realization, with its finite dimension packaged explicitly. -/
theorem posDef_exists_fd_independent_gram {n : ℕ}
    (B : Matrix (Fin n) (Fin n) ℝ) (hB : B.PosDef) :
    ∃ (E : Type) (_ : NormedAddCommGroup E) (_ : InnerProductSpace ℝ E)
      (_ : FiniteDimensional ℝ E) (u : Fin n → E),
      Matrix.gram ℝ u = B ∧ LinearIndependent ℝ u ∧ Module.finrank ℝ E = n := by
  letI := B.toNormedAddCommGroup hB
  letI := B.toInnerProductSpace hB.posSemidef
  let u : Fin n → (Fin n → ℝ) := fun i => Pi.single i 1
  have hgram : Matrix.gram ℝ u = B := by
    classical
    ext i j
    change ((B *ᵥ Pi.single j 1) ⬝ᵥ star (Pi.single i 1)) = B i j
    rw [Matrix.mulVec_single]
    simp only [MulOpposite.op_one, one_smul, dotProduct, Pi.star_apply,
      star_trivial, Matrix.col_apply]
    rw [Finset.sum_eq_single i]
    · simp
    · intro b hb hbi
      simp [hbi]
    · simp
  haveI : FiniteDimensional ℝ (Fin n → ℝ) := inferInstance
  have hfinrank : Module.finrank ℝ (Fin n → ℝ) = n := by simp
  exact ⟨Fin n → ℝ, inferInstance, inferInstance, inferInstance, u, hgram,
    Pi.linearIndependent_single_one (Fin n) ℝ, hfinrank⟩

/-- The squared norm of a finite block sum is the corresponding double Gram sum. -/
theorem norm_sq_block_sum_eq_gram_sum {E : Type*} [NormedAddCommGroup E]
    [InnerProductSpace ℝ E] (u : ℕ → E) (i j : ℕ) :
    ‖∑ a ∈ Finset.Ico i j, u a‖ ^ 2 =
      ∑ a ∈ Finset.Ico i j, ∑ b ∈ Finset.Ico i j, Matrix.gram ℝ u a b := by
  rw [← real_inner_self_eq_norm_sq]
  simp only [sum_inner, inner_sum, Matrix.gram_apply]
  apply Finset.sum_congr rfl
  intro a ha
  apply Finset.sum_congr rfl
  intro b hb
  exact real_inner_comm _ _

/-- Successive partial sums differ by precisely the intervening block of increments. -/
theorem partial_sum_sub_partial_sum {E : Type*} [AddCommGroup E]
    (u : ℕ → E) {i j : ℕ} (hij : i ≤ j) :
    (∑ a ∈ Finset.range j, u a) - (∑ a ∈ Finset.range i, u a) =
      ∑ a ∈ Finset.Ico i j, u a := by
  rw [Finset.sum_Ico_eq_sub u hij]

theorem prefixMatrix_det (n : ℕ) : (prefixMatrix n).det = 1 := by
  rw [Matrix.det_of_upperTriangular]
  · simp [prefixMatrix]
  · intro i j hji
    have hji' : j < i := by simpa using hji
    simp [prefixMatrix, not_le.mpr hji']

theorem prefixMatrix_columns_linearIndependent (n : ℕ) :
    LinearIndependent ℝ (prefixMatrix n)ᵀ := by
  apply Matrix.linearIndependent_cols_of_det_ne_zero
  rw [prefixMatrix_det]
  norm_num

/-- Prefix sums of a linearly independent finite family remain linearly independent. -/
theorem linearIndependent_prefixSums {n : ℕ} {E : Type*}
    [AddCommMonoid E] [Module ℝ E] (u : Fin n → E)
    (hu : LinearIndependent ℝ u) : LinearIndependent ℝ (prefixSums u) := by
  let L : (Fin n → ℝ) →ₗ[ℝ] E := Fintype.linearCombination ℝ u
  have hLinj : Function.Injective L := by
    exact linearIndependent_iff_injective_fintypeLinearCombination.mp hu
  have hmapped : LinearIndependent ℝ (L ∘ (prefixMatrix n)ᵀ) :=
    (prefixMatrix_columns_linearIndependent n).map_injOn L hLinj.injOn
  have heq : L ∘ (prefixMatrix n)ᵀ = prefixSums u := by
    funext j
    change Fintype.linearCombination ℝ u ((prefixMatrix n)ᵀ j) = prefixSums u j
    rw [Fintype.linearCombination_apply]
    simp only [transpose_apply, prefixMatrix, prefixSums]
    simpa only [ite_smul, one_smul, zero_smul] using
      (Finset.sum_filter (s := Finset.univ) (fun i : Fin n => i ≤ j) u).symm
  rwa [heq] at hmapped





/-- Cumulative points of independent increments are affinely independent. -/
theorem affineIndependent_cumulativePoints {n : ℕ} {E : Type*}
    [AddCommGroup E] [Module ℝ E] (u : Fin n → E)
    (hu : LinearIndependent ℝ u) : AffineIndependent ℝ (cumulativePoints u) := by
  rw [affineIndependent_iff_linearIndependent_vsub ℝ (cumulativePoints u) 0]
  rw [← linearIndependent_equiv (finSuccNeZeroEquiv n)]
  simpa [Function.comp_def, cumulativePoints, finSuccNeZeroEquiv] using
    linearIndependent_prefixSums u hu

/-- A positive-definite real matrix is the Gram matrix of independent standard Euclidean vectors. -/
theorem posDef_exists_euclidean_independent_gram {n : ℕ}
    (B : Matrix (Fin n) (Fin n) ℝ) (hB : B.PosDef) :
    ∃ v : Fin n → EuclideanSpace ℝ (Fin n),
      Matrix.gram ℝ v = B ∧ LinearIndependent ℝ v := by
  obtain ⟨E, instNorm, instInner, instFD, u, hGram, hLI, hdim⟩ :=
    posDef_exists_fd_independent_gram B hB
  letI := instNorm
  letI := instInner
  letI := instFD
  let b := (stdOrthonormalBasis ℝ E).reindex (finCongr hdim)
  let L := b.repr
  let v : Fin n → EuclideanSpace ℝ (Fin n) := fun i => L (u i)
  have hvGram : Matrix.gram ℝ v = B := by
    ext i j
    change inner ℝ (L (u i)) (L (u j)) = B i j
    exact (L.inner_map_map (u i) (u j)).trans (congrFun (congrFun hGram i) j)
  have hvLI : LinearIndependent ℝ v := by
    exact hLI.map_injOn L.toLinearMap L.injective.injOn
  exact ⟨v, hvGram, hvLI⟩



/-- A finite prefix sum equals the corresponding range sum of the zero extension. -/
theorem prefixSums_eq_sum_range {n : ℕ} {E : Type*} [AddCommMonoid E]
    (v : Fin n → E) (j : Fin n) :
    prefixSums v j = ∑ a ∈ Finset.range (j.val + 1), finExtendZero v a := by
  let e : Fin (j.val + 1) ≃ {i : Fin n // i.val < j.val + 1} :=
    Fin.castLEquiv (Nat.succ_le_iff.mpr j.isLt)
  have hsub :
      (∑ i : {i : Fin n // i.val < j.val + 1}, v i) =
        ∑ i : Fin (j.val + 1), v (e i) := by
    exact (Fintype.sum_equiv e (fun i => v (e i)) (fun i => v i) (fun _ => rfl)).symm
  rw [prefixSums]
  calc
    (∑ i : Fin n with i ≤ j, v i) =
        ∑ i : {i : Fin n // i.val < j.val + 1}, v i := by
          apply Finset.sum_subtype
          intro i
          simp only [Finset.mem_filter, Finset.mem_univ, true_and]
          omega
    _ = ∑ i : Fin (j.val + 1), v (e i) := hsub
    _ = ∑ a ∈ Finset.range (j.val + 1), finExtendZero v a := by
      rw [Finset.sum_fin_eq_sum_range]
      apply Finset.sum_congr rfl
      intro a ha
      have ha' : a < j.val + 1 := Finset.mem_range.mp ha
      have han : a < n := lt_of_lt_of_le ha' (Nat.succ_le_iff.mpr j.isLt)
      have haj : a ≤ j.val := by omega
      simp [e, finExtendZero, han, haj]

/-- Cumulative points are natural-number partial sums of the zero-extended increments. -/
theorem cumulativePoints_eq_sum_range {n : ℕ} {E : Type*} [AddCommMonoid E]
    (v : Fin n → E) (j : Fin (n + 1)) :
    cumulativePoints v j = ∑ a ∈ Finset.range j.val, finExtendZero v a := by
  refine Fin.cases ?_ (fun k => ?_) j
  · simp [cumulativePoints]
  · simpa [cumulativePoints] using prefixSums_eq_sum_range v k



theorem sqDistanceExtendZero_symmetric {n : ℕ} (D : SqDistanceData (Fin (n + 1)))
    (hD : D.IsHermitian) (a b : ℕ) :
    sqDistanceExtendZero D a b = sqDistanceExtendZero D b a := by
  by_cases ha : a < n + 1 <;> by_cases hb : b < n + 1
  · simp only [sqDistanceExtendZero, ha, hb, ↓reduceDIte]
    simpa only [star_trivial] using hD.apply ⟨b, hb⟩ ⟨a, ha⟩
  · simp [sqDistanceExtendZero, ha, hb]
  · simp [sqDistanceExtendZero, ha, hb]
  · simp [sqDistanceExtendZero, ha, hb]

theorem sqDistanceExtendZero_diag {n : ℕ} (D : SqDistanceData (Fin (n + 1)))
    (hdiag : ∀ i, D i i = 0) (a : ℕ) : sqDistanceExtendZero D a a = 0 := by
  by_cases ha : a < n + 1
  · simp [sqDistanceExtendZero, ha, hdiag]
  · simp [sqDistanceExtendZero, ha]

/-- Positive increment Gram data realize all ordered squared distances in standard Euclidean space. -/
theorem posDef_incrementGram_realizes_in_euclideanSpace_ordered {n : ℕ}
    (D : SqDistanceData (Fin (n + 1))) (hDsym : D.IsHermitian)
    (hdiag : ∀ i, D i i = 0) (hB : (incrementGram D).PosDef) :
    ∃ p : Fin (n + 1) → EuclideanSpace ℝ (Fin n),
      AffineIndependent ℝ p ∧
      ∀ i j : Fin (n + 1), i.val ≤ j.val → ‖p j - p i‖ ^ 2 = D i j := by
  obtain ⟨v, hvGram, hvLI⟩ :=
    posDef_exists_euclidean_independent_gram (incrementGram D) hB
  let p := cumulativePoints v
  refine ⟨p, affineIndependent_cumulativePoints v hvLI, ?_⟩
  intro i j hij
  rw [show p j = ∑ a ∈ Finset.range j.val, finExtendZero v a from
        cumulativePoints_eq_sum_range v j,
      show p i = ∑ a ∈ Finset.range i.val, finExtendZero v a from
        cumulativePoints_eq_sum_range v i,
      partial_sum_sub_partial_sum (finExtendZero v) hij,
      norm_sq_block_sum_eq_gram_sum]
  have hsum :
      (∑ a ∈ Finset.Ico i.val j.val, ∑ b ∈ Finset.Ico i.val j.val,
        Matrix.gram ℝ (finExtendZero v) a b) =
      ∑ a ∈ Finset.Ico i.val j.val, ∑ b ∈ Finset.Ico i.val j.val,
        (-(1 / 2 : ℝ) *
          (sqDistanceExtendZero D (a + 1) (b + 1) -
           sqDistanceExtendZero D (a + 1) b -
           sqDistanceExtendZero D a (b + 1) +
           sqDistanceExtendZero D a b)) := by
    apply Finset.sum_congr rfl
    intro a ha
    have hai : i.val ≤ a := (Finset.mem_Ico.mp ha).1
    have haj : a < j.val := (Finset.mem_Ico.mp ha).2
    have han : a < n := lt_of_lt_of_le haj (Nat.lt_succ_iff.mp j.isLt)
    apply Finset.sum_congr rfl
    intro b hb
    have hbj : b < j.val := (Finset.mem_Ico.mp hb).2
    have hbn : b < n := lt_of_lt_of_le hbj (Nat.lt_succ_iff.mp j.isLt)
    have hgramab := congrFun (congrFun hvGram ⟨a, han⟩) ⟨b, hbn⟩
    simp only [Matrix.gram_apply, finExtendZero, han, hbn, ↓reduceDIte] at hgramab ⊢
    rw [hgramab]
    simp [incrementGram, sqDistanceExtendZero, han, hbn, Nat.lt_succ_of_lt han,
      Nat.lt_succ_of_lt hbn]
  rw [hsum]
  rw [mixed_difference_block_eq_distance (sqDistanceExtendZero D)
    (sqDistanceExtendZero_symmetric D hDsym)
    (sqDistanceExtendZero_diag D hdiag) hij]
  simp [sqDistanceExtendZero, i.isLt, j.isLt]

/-- Positive increment Gram data realize all squared distances in standard Euclidean space. -/
theorem posDef_incrementGram_realizes_in_euclideanSpace {n : ℕ}
    (D : SqDistanceData (Fin (n + 1))) (hDsym : D.IsHermitian)
    (hdiag : ∀ i, D i i = 0) (hB : (incrementGram D).PosDef) :
    ∃ p : Fin (n + 1) → EuclideanSpace ℝ (Fin n),
      AffineIndependent ℝ p ∧ ∀ i j, ‖p j - p i‖ ^ 2 = D i j := by
  obtain ⟨p, hpaff, hp⟩ :=
    posDef_incrementGram_realizes_in_euclideanSpace_ordered D hDsym hdiag hB
  refine ⟨p, hpaff, ?_⟩
  intro i j
  by_cases hij : i.val ≤ j.val
  · exact hp i j hij
  · have hji : j.val ≤ i.val := Nat.le_of_lt (Nat.lt_of_not_ge hij)
    have h := hp j i hji
    have hnorm : ‖p j - p i‖ = ‖p i - p j‖ := by
      rw [show p j - p i = -(p i - p j) by abel, norm_neg]
    have hD : D j i = D i j := by
      simpa only [star_trivial] using hDsym.apply i j
    rw [hnorm, h, hD]

end Erdos488.Geometry.SeparatedDistances

/-- Pure-colex prescribed distances are symmetric. -/
theorem colexDistance_symm {m : ℕ} (K : ℝ) (i j : Fin (m + 1)) :
    colexDistance K i j = colexDistance K j i := by
  by_cases hij : i.val < j.val
  · simp [colexDistance, hij, Nat.not_lt.mpr hij.le]

    · simp [colexDistance, hji, Nat.not_lt.mpr hji.le]
    · have h : i = j := Fin.ext (Nat.le_antisymm (Nat.not_lt.mp hji) (Nat.not_lt.mp hij))
      subst j
      simp

/-- Every pure-colex comparison weight is positive once `K ≥ 10`. -/
theorem colexWeight_pos {K : ℝ} (hK : 10 ≤ K) (j : ℕ) (hj : 1 ≤ j) :
    0 < colexWeight K j := by
  unfold colexWeight
  split_ifs with h
  · have hKpos : 0 < K := by linarith
    positivity
  · have hjR : (0 : ℝ) < j := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one hj)
    have hKpos : 0 < K := by linarith
    positivity

/-- The pure-colex comparison form is nonnegative. -/
theorem colexQ_nonneg {m : ℕ} {K : ℝ} (hK : 10 ≤ K) (x : Fin m → ℝ) :
    0 ≤ colexQ K x := by
  unfold colexQ
  apply Finset.sum_nonneg
  intro i hi
  exact mul_nonneg (colexWeight_pos hK _ (by omega)).le (sq_nonneg _)

/-- The pure-colex comparison form is strictly positive away from zero. -/
theorem colexQ_pos {m : ℕ} {K : ℝ} (hK : 10 ≤ K) {x : Fin m → ℝ}
    (hx : x ≠ 0) : 0 < colexQ K x := by
  obtain ⟨i, hi⟩ := Function.ne_iff.mp hx
  unfold colexQ
  apply Finset.sum_pos'
  · intro j hj
    exact mul_nonneg (colexWeight_pos hK _ (by omega)).le (sq_nonneg _)
  · refine ⟨i, Finset.mem_univ _, ?_⟩
    exact mul_pos (colexWeight_pos hK _ (by omega)) (sq_pos_of_ne_zero hi)













/-- Weighted Cauchy bound for the sum of absolute coordinates. -/
theorem colex_abs_sum_sq_le {m : ℕ} {K : ℝ} (hK : 10 ≤ K) (x : Fin m → ℝ) :
    (∑ i, |x i|) ^ 2 ≤ colexS0 (m := m) K * colexQ K x := by
  unfold colexS0 colexQ
  apply Finset.sum_sq_le_sum_mul_sum_of_sq_le_mul Finset.univ
  · intro i hi

/-- The real harmonic number is at least one once its index is nonzero. -/
theorem one_le_harmonic_of_one_le {n : ℕ} (hn : 1 ≤ n) :
    (1 : ℝ) ≤ (harmonic n : ℝ) := by

/-- A uniform reciprocal-weight bound, deliberately coarse at the first two
coordinates so that no finite-sum split is needed. -/
theorem colexWeight_inv_le {K : ℝ} (hK : 10 ≤ K) {j : ℕ} (hj : 1 ≤ j) :
    (colexWeight K j)⁻¹ ≤ 29 / (K * j) := by
  have hKpos : 0 < K := by linarith
  have hjR : (0 : ℝ) < j := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one hj)
  unfold colexWeight
  split_ifs with hsmall
  · have hj2R : (j : ℝ) ≤ 2 := by exact_mod_cast hsmall
    rw [inv_div]
    apply (le_div_iff₀ (mul_pos hKpos hjR)).2
    rw [div_mul_eq_mul_div]
    apply (div_le_iff₀ (sq_pos_of_pos hKpos)).2
    nlinarith
  · have hden : 0 < K * (j : ℝ) + (j : ℝ) ^ 3 := by positivity
    rw [inv_div]
    apply (le_div_iff₀ (mul_pos hKpos hjR)).2
    rw [div_mul_eq_mul_div]
    apply (div_le_iff₀ hden).2
    nlinarith [sq_nonneg (j : ℝ)]

/-- A uniform indexed reciprocal-weight bound. -/
theorem colexIndexSq_div_weight_le {K : ℝ} (hK : 10 ≤ K) {j : ℕ}

  have hKpos : 0 < K := by linarith
  have hjR : (0 : ℝ) < j := by exact_mod_cast (lt_of_lt_of_le Nat.zero_lt_one hj)

  split_ifs with hsmall
  · have hj2R : (j : ℝ) ≤ 2 := by exact_mod_cast hsmall
    apply (le_div_iff₀ hjR).2

    apply (div_le_iff₀ (div_pos (sq_pos_of_pos hKpos) (by norm_num : (0 : ℝ) < 24))).2
    nlinarith [sq_nonneg (j : ℝ), mul_self_nonneg ((j : ℝ) - 1)]
  · have hden : 0 < (K * (j : ℝ) + (j : ℝ) ^ 3) / 24 := by positivity
    apply (le_div_iff₀ hjR).2
    rw [div_mul_eq_mul_div]
    apply (div_le_iff₀ hden).2
    nlinarith [sq_nonneg (j : ℝ)]

/-- The reciprocal one-based indices carried by `Fin m` form an initial
segment of the real harmonic sum through `m+1`. -/
theorem fin_reciprocal_sum_le_harmonic_succ (m : ℕ) :
    (∑ i : Fin m, ((i.val + 1 : ℝ)⁻¹)) ≤ (harmonic (m + 1) : ℝ) := by
  rw [harmonic]
  push_cast
  have heq : (∑ i : Fin m, ((i.val + 1 : ℝ)⁻¹)) =
      ∑ i ∈ Finset.range m, ((i + 1 : ℝ)⁻¹) := by
    rw [Finset.sum_fin_eq_sum_range]
    apply Finset.sum_congr rfl
    intro i hi
    simp [Finset.mem_range.mp hi]
  rw [heq]
  apply Finset.sum_le_sum_of_subset_of_nonneg (Finset.range_mono (Nat.le_succ m))
  intro i hi hnot
  positivity

/-- Coarse harmonic control of the reciprocal comparison weights. -/
theorem colex_S0_le {m : ℕ} {K : ℝ} (hK : 10 ≤ K) :
    colexS0 (m := m) K ≤ 29 * (harmonic (m + 1) : ℝ) / K := by
  have hKpos : 0 < K := by linarith
  unfold colexS0
  calc
    ∑ i : Fin m, (colexWeight K (i.val + 1))⁻¹ ≤
        ∑ i : Fin m, 29 / (K * (i.val + 1 : ℝ)) := by
          apply Finset.sum_le_sum
          intro i hi
          simpa using colexWeight_inv_le hK (j := i.val + 1)
            (Nat.succ_le_succ (Nat.zero_le i.val))
    _ = (29 / K) * ∑ i : Fin m, ((i.val + 1 : ℝ)⁻¹) := by
          rw [Finset.mul_sum]
          apply Finset.sum_congr rfl
          intro i hi
          have hj : (0 : ℝ) < (i.val + 1 : ℕ) := by positivity
          field_simp [hKpos.ne', hj.ne']
    _ ≤ (29 / K) * (harmonic (m + 1) : ℝ) := by
          exact mul_le_mul_of_nonneg_left (fin_reciprocal_sum_le_harmonic_succ m)
            (div_nonneg (by norm_num) hKpos.le)
    _ = 29 * (harmonic (m + 1) : ℝ) / K := by ring

/-- Coarse harmonic control of the indexed reciprocal comparison weights. -/
theorem colex_S2_le {m : ℕ} {K : ℝ} (hK : 10 ≤ K) :

    ∑ i : Fin m, ((i.val + 1 : ℝ) ^ 2 / colexWeight K (i.val + 1)) ≤
        ∑ i : Fin m, 26 / (i.val + 1 : ℝ) := by

          rw [Finset.mul_sum]
          apply Finset.sum_congr rfl
          intro i hi
          rw [div_eq_mul_inv]
    _ ≤ 26 * (harmonic (m + 1) : ℝ) := by
          exact mul_le_mul_of_nonneg_left (fin_reciprocal_sum_le_harmonic_succ m)
            (by norm_num)

private lemma two_product_le_two_sqrt_product (a b s t q : ℝ)
    (hs : 0 ≤ s) (ht : 0 ≤ t) (hq : 0 ≤ q)
    (ha2 : a ^ 2 ≤ s * q) (hb2 : b ^ 2 ≤ t * q) :
    2 * a * b ≤ 2 * Real.sqrt (s * t) * q := by
  have hst : 0 ≤ s * t := mul_nonneg hs ht
  have hsq : (Real.sqrt (s * t)) ^ 2 = s * t := Real.sq_sqrt hst
  have hroot : 0 ≤ Real.sqrt (s * t) := Real.sqrt_nonneg _
  have hsq1 := mul_le_mul_of_nonneg_right ha2 (sq_nonneg b)
  have hsq2 := mul_le_mul_of_nonneg_left hb2 (mul_nonneg hs hq)
  have hprod : (a * b) ^ 2 ≤ (Real.sqrt (s * t) * q) ^ 2 := by
    calc
      (a * b) ^ 2 = a ^ 2 * b ^ 2 := by ring
      _ ≤ s * q * b ^ 2 := hsq1
      _ ≤ s * q * (t * q) := hsq2
      _ = (s * t) * q ^ 2 := by ring
      _ = (Real.sqrt (s * t)) ^ 2 * q ^ 2 := by
        exact (congrArg (fun z : ℝ => z * q ^ 2) hsq).symm
      _ = (Real.sqrt (s * t) * q) ^ 2 := by ring
  nlinarith [mul_nonneg hroot hq]

private lemma sqrt_product_budget (s t H K : ℝ)
    (_hs : 0 ≤ s) (ht : 0 ≤ t) (hH : 0 ≤ H) (hK : 0 < K)
    (hlarge : 16384 * H ^ 2 ≤ K)
    (hsb : s ≤ 29 * H / K) (htb : t ≤ 26 * H) :
    2 * Real.sqrt (s * t) ≤ Real.sqrt (754 : ℝ) / 64 := by
  have hprod : s * t ≤ (754 / 16384 : ℝ) := by
    calc
      s * t ≤ (29 * H / K) * t := mul_le_mul_of_nonneg_right hsb ht
      _ ≤ (29 * H / K) * (26 * H) :=
        mul_le_mul_of_nonneg_left htb (div_nonneg (by positivity) hK.le)
      _ = 754 * H ^ 2 / K := by ring
      _ ≤ (754 / 16384 : ℝ) := by
        apply (div_le_iff₀ hK).2
        nlinarith [hlarge]
  have hsqrt := Real.sqrt_le_sqrt hprod
  have h754 : (Real.sqrt (754 : ℝ)) ^ 2 = 754 := Real.sq_sqrt (by norm_num)
  have hquot : (Real.sqrt (754 / 16384 : ℝ)) ^ 2 = 754 / 16384 :=
    Real.sq_sqrt (by norm_num)
  have hroot_eq : Real.sqrt (754 / 16384 : ℝ) = Real.sqrt (754 : ℝ) / 128 := by
    nlinarith [Real.sqrt_nonneg (754 / 16384 : ℝ), Real.sqrt_nonneg (754 : ℝ)]
  rw [hroot_eq] at hsqrt
  linarith

private lemma sqrt754_div64_lt_half : Real.sqrt (754 : ℝ) / 64 < 1 / 2 := by
  have hsq : (Real.sqrt (754 : ℝ)) ^ 2 = 754 := Real.sq_sqrt (by norm_num)
  have hnonneg : 0 ≤ Real.sqrt (754 : ℝ) := Real.sqrt_nonneg _
  nlinarith

/-- Direct Cauchy--Schwarz on the actual colex remainder and its two
reciprocal-weight sums. The intermediate radical is retained explicitly. -/
theorem colex_remainder_abs_le_direct_cauchy {m : ℕ} (hm : 2 ≤ m)
    {K : ℝ} (hK : 10 ≤ K)
    (hlarge : 16384 * (harmonic (m + 1) : ℝ) ^ 2 ≤ K) (x : Fin m → ℝ) :
    |star x ⬝ᵥ (colexRemainder K *ᵥ x)| ≤
      2 * Real.sqrt (colexS0 (m := m) K * colexS2 (m := m) K) * colexQ K x ∧
    2 * Real.sqrt (colexS0 (m := m) K * colexS2 (m := m) K) * colexQ K x ≤
      Real.sqrt (754 : ℝ) / 64 * colexQ K x := by
  let a : ℝ := ∑ i, |x i|
  let b : ℝ := ∑ i, (i.val + 1 : ℝ) * |x i|
  let q : ℝ := colexQ K x
  let H : ℝ := (harmonic (m + 1) : ℝ)
  have hq : 0 ≤ q := colexQ_nonneg hK x
  have hs : 0 ≤ colexS0 (m := m) K := by
    unfold colexS0
    apply Finset.sum_nonneg
    intro i hi
    exact inv_nonneg.mpr (colexWeight_pos hK _ (by omega)).le
  have ht : 0 ≤ colexS2 (m := m) K := by
    unfold colexS2
    apply Finset.sum_nonneg
    intro i hi
    exact div_nonneg (sq_nonneg _) (colexWeight_pos hK _ (by omega)).le
  have ha2 : a ^ 2 ≤ colexS0 (m := m) K * q := by
    simpa [a, q] using colex_abs_sum_sq_le hK x
  have hb2 : b ^ 2 ≤ colexS2 (m := m) K * q := by
    simpa [b, q] using colex_indexed_abs_sum_sq_le hK x
  have hH1 : 1 ≤ H := by
    simpa [H] using (one_le_harmonic_of_one_le (by omega : 1 ≤ m + 1))
  have hH : 0 ≤ H := by linarith
  have hKpos : 0 < K := by linarith
  have hsbudget : colexS0 (m := m) K ≤ 29 * H / K := by
    simpa [H] using (colex_S0_le (m := m) hK)
  have htbudget : colexS2 (m := m) K ≤ 26 * H := by
    simpa [H] using (colex_S2_le (m := m) hK)
  have hCS := two_product_le_two_sqrt_product a b
    (colexS0 (m := m) K) (colexS2 (m := m) K) q
    hs ht hq ha2 hb2
  have hcoef := sqrt_product_budget (colexS0 (m := m) K)
    (colexS2 (m := m) K) H K hs ht hH hKpos hlarge hsbudget htbudget
  constructor
  · calc
      |star x ⬝ᵥ (colexRemainder K *ᵥ x)| ≤ 2 * a * b := by
        simpa [a, b] using colex_remainder_quadratic_abs_le_product K x
      _ ≤ 2 * Real.sqrt (colexS0 (m := m) K * colexS2 (m := m) K) * q := hCS
      _ = 2 * Real.sqrt (colexS0 (m := m) K * colexS2 (m := m) K) *
          colexQ K x := rfl
  · exact mul_le_mul_of_nonneg_right hcoef hq

theorem colex_incrementGram_ge_half_tridiagonal_direct {m : ℕ} (hm : 2 ≤ m)

  simp only [Matrix.sub_mulVec, dotProduct_sub]
  linarith

    rw [← Matrix.mulVec_mulVec, Matrix.dotProduct_mulVec]
    rw [Matrix.vecMul_transpose]
    simp only [dotProduct, pow_two]

lemma weighted_duality_sq {ι : Type*} [Fintype ι]
    (P u z : ι → ℝ) (hP : ∀ i, P i ≠ 0) :
    (u ⬝ᵥ z)^2 ≤ (∑ i, P i ^ 2 * u i ^ 2) * ∑ i, (z i / P i)^2 := by
  have h := Finset.sum_mul_sq_le_sq_mul_sq Finset.univ
    (fun i => P i * u i) (fun i => z i / P i)
  have hh : ∀ i, (P i * u i) * (z i / P i) = u i * z i := by
    intro i
    field_simp [hP i]
  simpa only [hh, mul_pow, dotProduct] using h




/-- Arbitrary edge-potential energy bound. Since v may be any linear combination
of columns, this is the uniform quadratic-form input to the whole-operator
estimate, not an average or columnwise estimate. -/
theorem flattened_colex_edge_solve_energy {m : ℕ} (hm : 1 ≤ m)
    {K : ℝ} (hK : 10 ≤ K)
    (hlarge : 16384 * (harmonic (m+2) : ℝ)^2 ≤ K)
    (r : ℝ) (hr : r ≤ K^2/16) (x v : Fin (m+1) → ℝ)
    (hsolve : incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r) *ᵥ x = v) :
    x ⬝ᵥ (incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r) *ᵥ x) ≤
      144*(K+(m+1:ℕ)) * ∑ i, (v i / colexANat K (i.val+1))^2 := by
  let A := incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r)
  let q := x ⬝ᵥ (A *ᵥ x)
  let C := ∑ i, (v i / colexANat K (i.val+1))^2
  have hA : A.PosDef := flattened_colex_posDef hm hK hlarge r hr
  have hq0 : 0 ≤ q := by
    simpa only [q, star_trivial] using hA.posSemidef.dotProduct_mulVec_nonneg x
  have hC : 0 ≤ C := Finset.sum_nonneg fun i _ => sq_nonneg _
  have hP : ∀ i : Fin (m+1), colexANat K (i.val+1) ≠ 0 := by
    intro i
    unfold colexANat
    have hn : (0:ℝ) ≤ Nat.choose (i.val+1+1) 2 := by positivity
    linarith
  have hduality := weighted_duality_sq (fun i => colexANat K (i.val+1)) x v hP
  have hqv : q = x ⬝ᵥ v := by dsimp [q,A]; rw [hsolve]
  have hreserve := flattened_colex_diagonal_reserve hm hK hlarge r hr x
  have hbound := mul_le_mul_of_nonneg_right hreserve hC
  have hsq : q*q ≤ q*(144*(K+(m+1:ℕ))*C) := by
    rw [← hqv] at hduality
    dsimp [q,A,C] at *
    nlinarith
  change q ≤ 144*(K+(m+1:ℕ))*C
  rcases eq_or_lt_of_le hq0 with hzero | hpos
  · rw [← hzero]; positivity
  · exact le_of_mul_le_mul_left hsq hpos

/-- Whole coefficient-matrix version: the inequality holds for every vector
of column coefficients, for the single linear solution X=A^{-1}Z. -/
theorem flattened_colex_whole_solve_energy {m : ℕ} (hm : 1 ≤ m)
    {K : ℝ} (hK : 10 ≤ K)
    (hlarge : 16384 * (harmonic (m+2) : ℝ)^2 ≤ K)
    (r : ℝ) (hr : r ≤ K^2/16)
    {σ : Type*} [Fintype σ] (Z : Matrix (Fin (m+1)) σ ℝ) :
    ∃ X : Matrix (Fin (m+1)) σ ℝ,
      incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r) * X = Z ∧
      ∀ t : σ → ℝ,
        (X *ᵥ t) ⬝ᵥ (incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r) *ᵥ (X *ᵥ t)) ≤
          144*(K+(m+1:ℕ)) * ∑ i, ((Z *ᵥ t) i / colexANat K (i.val+1))^2 := by
  let A := incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r)
  have hA : A.PosDef := flattened_colex_posDef hm hK hlarge r hr
  have hdet : IsUnit A.det := (Matrix.isUnit_iff_isUnit_det A).mp hA.isUnit
  let X := A⁻¹ * Z
  have hX : A*X=Z := by
    dsimp [X]
    rw [← Matrix.mul_assoc, Matrix.mul_nonsing_inv A hdet, Matrix.one_mul]
  refine ⟨X, hX, ?_⟩
  intro t
  apply flattened_colex_edge_solve_energy hm hK hlarge r hr
  change A *ᵥ (X *ᵥ t) = Z *ᵥ t
  rw [Matrix.mulVec_mulVec, hX]

lemma nonnegative_weighted_cauchy {ι : Type*} [Fintype ι]
    (a w x : ι → ℝ) (ha : ∀ i, 0 ≤ a i) (hw : ∀ i, 0 < w i) :
    (∑ i, a i*x i)^2 ≤ (∑ i, a i*w i) * ∑ i, a i/w i*x i^2 := by
  have h := Finset.sum_mul_sq_le_sq_mul_sq Finset.univ
    (fun i => Real.sqrt (a i)*Real.sqrt (w i))
    (fun i => Real.sqrt (a i)/Real.sqrt (w i)*x i)
  have hs : ∀ i, Real.sqrt (w i) ≠ 0 := fun i => ne_of_gt (Real.sqrt_pos.2 (hw i))
  have hprod : ∀ i, (Real.sqrt (a i)*Real.sqrt (w i)) *
      (Real.sqrt (a i)/Real.sqrt (w i)*x i) = a i*x i := by
    intro i
    have hh := Real.sq_sqrt (ha i)
    field_simp [hs i]
    rw [hh]
    ring
  have hf : ∀ i, (Real.sqrt (a i)*Real.sqrt (w i))^2 = a i*w i := by
    intro i
    rw [mul_pow, Real.sq_sqrt (ha i), Real.sq_sqrt (hw i).le]
  have hg : ∀ i, (Real.sqrt (a i)/Real.sqrt (w i)*x i)^2 = a i/w i*x i^2 := by
    intro i
    rw [mul_pow, div_pow, Real.sq_sqrt (ha i), Real.sq_sqrt (hw i).le]
  simpa only [hprod,hf,hg] using h

/-- Finite weighted Schur test at the Hardy constant two, stated as a
simultaneous quadratic-form bound for every input vector. -/
theorem weighted_schur_two {ι : Type*} [Fintype ι]
    (A : Matrix ι ι ℝ) (w : ι → ℝ)
    (hA : ∀ i j, 0 ≤ A i j) (hw : ∀ i, 0 < w i)
    (hrow : ∀ i, ∑ j, A i j*w j ≤ 2*w i)
    (hcol : ∀ j, ∑ i, A i j*w i ≤ 2*w j)
    (x : ι → ℝ) :
    (∑ i, (A *ᵥ x) i ^ 2) ≤ 4*∑ j, x j ^ 2 := by
  have hlocal : ∀ i, (A *ᵥ x) i ^ 2 ≤ 2*w i * ∑ j, A i j/w j*x j^2 := by
    intro i
    have hc := nonnegative_weighted_cauchy (A i) w x (hA i) hw
    have hz : 0 ≤ ∑ j, A i j/w j*x j^2 := by
      apply Finset.sum_nonneg
      intro j hj
      exact mul_nonneg (div_nonneg (hA i j) (hw j).le) (sq_nonneg _)
    have hh := mul_le_mul_of_nonneg_right (hrow i) hz
    simpa only [Matrix.mulVec, dotProduct] using hc.trans hh
  calc
    _ ≤ ∑ i, 2*w i * ∑ j, A i j/w j*x j^2 :=
      Finset.sum_le_sum fun i _ => hlocal i
    _ = ∑ j, (2*(∑ i, A i j*w i)/w j)*x j^2 := by
      simp only [Finset.mul_sum]
      rw [Finset.sum_comm]
      apply Finset.sum_congr rfl
      intro j hj
      simp only [Finset.sum_div, Finset.sum_mul, Finset.mul_sum]
      apply Finset.sum_congr rfl
      intro i hi
      ring
    _ ≤ ∑ j, 4*x j^2 := by
      apply Finset.sum_le_sum
      intro j hj
      apply mul_le_mul_of_nonneg_right _ (sq_nonneg _)
      apply (div_le_iff₀ (hw j)).2
      linarith [hcol j]
    _ = _ := by rw [Finset.mul_sum]

lemma sqrt_recip_increment {x : ℝ} (hx : 1 ≤ x) :
    1/Real.sqrt x ≤ 2*(Real.sqrt x-Real.sqrt (x-1)) := by
  have hs : 0 < Real.sqrt x := Real.sqrt_pos.2 (by linarith)
  have h₁ := Real.sq_sqrt (show 0 ≤ x by linarith)
  have h₂ := Real.sq_sqrt (show 0 ≤ x-1 by linarith)
  apply (div_le_iff₀ hs).2
  nlinarith [sq_nonneg (Real.sqrt x-Real.sqrt (x-1))]

/-- Exact telescoping row estimate for the Hardy Schur weight. -/
theorem sqrt_recip_prefix_sum (n : ℕ) :
    (∑ j ∈ Finset.range n, 1/Real.sqrt (j+1 : ℕ)) ≤ 2*Real.sqrt n := by
  induction n with
  | zero => simp
  | succ n ih =>
      rw [Finset.sum_range_succ]
      have hp := sqrt_recip_increment (x:=(n+1:ℕ)) (by exact_mod_cast Nat.succ_le_succ (Nat.zero_le n))
      norm_num only [Nat.cast_add, Nat.cast_one, add_sub_cancel_right] at hp ih ⊢
      linarith

lemma sqrt_recip_tail_increment {x : ℝ} (hx : 1 ≤ x) :
    1/((x+1)*Real.sqrt (x+1)) ≤
      2*(1/Real.sqrt x-1/Real.sqrt (x+1)) := by
  let u := Real.sqrt x
  let v := Real.sqrt (x+1)
  have hu : 0 < u := Real.sqrt_pos.2 (by linarith)
  have hv : 0 < v := Real.sqrt_pos.2 (by linarith)
  have hu2 : u^2=x := Real.sq_sqrt (by linarith)
  have hv2 : v^2=x+1 := Real.sq_sqrt (by linarith)
  have hdiff : v^2-u^2=1 := by linarith
  have hmul := congrArg (fun t : ℝ => u*t) hdiff
  have hn := mul_nonneg (sq_nonneg (v-u)) (show 0 ≤ 2*v+u by linarith)
  have hcore : u ≤ 2*(v-u)*v^2 := by nlinarith
  have hfrac : 1/(v^2*v) ≤ 2*(v-u)/(u*v) := by
    apply (div_le_div_iff₀ (mul_pos (sq_pos_of_pos hv) hv) (mul_pos hu hv)).2
    have hm := mul_le_mul_of_nonneg_right hcore hv.le
    nlinarith
  have heq : 2*(v-u)/(u*v) = 2*(1/u-1/v) := by field_simp
  change 1/((x+1)*v) ≤ 2*(1/u-1/v)
  rw [← hv2, ← heq]
  exact hfrac

/-- Exact telescoping finite tail estimate, with the terminal correction kept. -/
theorem sqrt_recip_tail_sum (j n : ℕ) (hj : 1 ≤ j) :
    (∑ a ∈ Finset.range n,
      1/((j+a+1 : ℕ)*Real.sqrt (j+a+1 : ℕ))) ≤
      2*(1/Real.sqrt j-1/Real.sqrt (j+n : ℕ)) := by
  induction n with
  | zero => simp
  | succ n ih =>
      rw [Finset.sum_range_succ]
      have hp := sqrt_recip_tail_increment (x:=(j+n:ℕ)) (by exact_mod_cast (show 1 ≤ j+n by omega))
      norm_num only [Nat.cast_add, Nat.cast_one, Nat.succ_eq_add_one, add_assoc] at hp ih ⊢
      linarith

lemma sum_range_prefix_indicator (g : ℕ → ℝ) {i n : ℕ} (hi : i ≤ n) :
    (∑ j ∈ Finset.range n, if j < i then g j else 0) = ∑ j ∈ Finset.range i, g j := by
  rw [← Finset.sum_range_add_sum_Ico (fun j => if j < i then g j else 0) hi]
  have h₁ : (∑ j ∈ Finset.range i, if j < i then g j else 0) = ∑ j ∈ Finset.range i, g j := by
    apply Finset.sum_congr rfl
    intro j hj
    simp [Finset.mem_range.mp hj]
  have h₂ : (∑ j ∈ Finset.Ico i n, if j < i then g j else 0) = 0 := by
    apply Finset.sum_eq_zero
    intro j hj
    simp [not_lt.mpr (Finset.mem_Ico.mp hj).1]
  rw [h₁,h₂,add_zero]

lemma sum_range_suffix_indicator (g : ℕ → ℝ) {i n : ℕ} (hi : i ≤ n) :
    (∑ j ∈ Finset.range n, if i ≤ j then g j else 0) = ∑ j ∈ Finset.Ico i n, g j := by
  rw [← Finset.sum_range_add_sum_Ico (fun j => if i ≤ j then g j else 0) hi]
  have h₁ : (∑ j ∈ Finset.range i, if i ≤ j then g j else 0) = 0 := by
    apply Finset.sum_eq_zero
    intro j hj
    simp [not_le.mpr (Finset.mem_range.mp hj)]
  have h₂ : (∑ j ∈ Finset.Ico i n, if i ≤ j then g j else 0) = ∑ j ∈ Finset.Ico i n, g j := by
    apply Finset.sum_congr rfl
    intro j hj
    simp [(Finset.mem_Ico.mp hj).1]
  rw [h₁,h₂,zero_add]





lemma hardyWeight_pos {n : ℕ} (i : Fin n) : 0 < hardyWeight i := by

  positivity

lemma hardyMatrix_row {n : ℕ} (i : Fin n) :
    ∑ j, hardyMatrix n i j * hardyWeight j ≤ 2*hardyWeight i := by
  have heq : (∑ j, hardyMatrix n i j * hardyWeight j) =
      (∑ j ∈ Finset.range i.val, 1/Real.sqrt (j+1 : ℕ))/(i.val+1 : ℕ) := by
    simp only [hardyMatrix,hardyWeight,ite_mul,zero_mul]
    rw [Fin.sum_univ_eq_sum_range (fun j => if j < i.val then 1/(i.val+1:ℕ)*(1/Real.sqrt (j+1:ℕ)) else 0)]
    have hr := sum_range_prefix_indicator (fun j => 1/(i.val+1 : ℕ)*(1/Real.sqrt (j+1 : ℕ))) i.isLt.le
    rw [hr]
    rw [← Finset.mul_sum]
    ring
  rw [heq]
  have hp := sqrt_recip_prefix_sum i.val
  have hm : Real.sqrt (i.val : ℝ) ≤ Real.sqrt (i.val+1 : ℕ) := by
    apply Real.sqrt_le_sqrt
    exact_mod_cast Nat.le_succ i.val
  apply (div_le_iff₀ (show (0:ℝ)<(i.val+1:ℕ) by positivity)).2
  have hs := Real.sq_sqrt (show (0:ℝ)≤(i.val+1:ℕ) by positivity)
  have hz : Real.sqrt (i.val+1:ℕ) ≠ 0 := ne_of_gt (by positivity)
  have hx : 2*hardyWeight i*(i.val+1:ℕ) = 2*Real.sqrt (i.val+1:ℕ) := by
    unfold hardyWeight
    field_simp [hz]
    nlinarith [hs]
  rw [hx]
  linarith

lemma hardyMatrix_column {n : ℕ} (j : Fin n) :
    ∑ i, hardyMatrix n i j * hardyWeight i ≤ 2*hardyWeight j := by
  have heq : (∑ i, hardyMatrix n i j * hardyWeight i) =
      ∑ i ∈ Finset.Ico (j.val+1) n, 1/((i+1:ℕ)*Real.sqrt (i+1:ℕ)) := by
    simp only [hardyMatrix,hardyWeight,ite_mul,zero_mul]
    rw [Fin.sum_univ_eq_sum_range (fun i => if j.val < i then 1/(i+1:ℕ)*(1/Real.sqrt (i+1:ℕ)) else 0)]
    have hterm : ∀ i : ℕ, (if j.val < i then 1/(i+1:ℕ)*(1/Real.sqrt (i+1:ℕ)) else 0) =
      (if j.val+1 ≤ i then 1/((i+1:ℕ)*Real.sqrt (i+1:ℕ)) else 0) := by
      intro i
      simp only [Nat.lt_iff_add_one_le,one_div_mul_one_div]
    simp_rw [hterm]
    exact sum_range_suffix_indicator _ j.isLt
  rw [heq, Finset.sum_Ico_eq_sum_range]
  have ht := sqrt_recip_tail_sum (j.val+1) (n-(j.val+1)) (by omega)
  have ht' : (∑ a ∈ Finset.range (n-(j.val+1)),
      1/((j.val+1+a+1:ℕ)*Real.sqrt (j.val+1+a+1:ℕ))) ≤ 2/Real.sqrt (j.val+1:ℕ) := by
    have hnon : 0 ≤ 1/Real.sqrt (j.val+1+(n-(j.val+1)):ℕ) := by positivity
    rw [div_eq_mul_one_div]
    linarith
  simpa [hardyWeight,div_eq_mul_inv] using ht'

lemma gram_increments_of_sq_distances {n : ℕ} {E : Type*}

    (p : Fin (n+1) → E) (D : Matrix (Fin (n+1)) (Fin (n+1)) ℝ)
    (hD : ∀ i j, ‖p j-p i‖^2 = D i j) :
    Matrix.gram ℝ (fun i : Fin n => p i.succ-p i.castSucc) = incrementGram D := by
  ext i j
  have h11 := hD i.succ j.succ
  have h10 := hD i.succ j.castSucc
  have h01 := hD i.castSucc j.succ
  have h00 := hD i.castSucc j.castSucc
  simp only [norm_sub_sq_real] at h11 h10 h01 h00
  simp only [Matrix.gram_apply, incrementGram, inner_sub_left, inner_sub_right]
  nlinarith [real_inner_comm (p i.succ) (p j.succ),
    real_inner_comm (p i.succ) (p j.castSucc),
    real_inner_comm (p i.castSucc) (p j.succ),
    real_inner_comm (p i.castSucc) (p j.castSucc)]

    (pathIncidence n *ᵥ z) i = z i.castSucc-z i.succ := by
  simp [pathIncidence, Matrix.mulVec, dotProduct, sub_mul, Finset.sum_sub_distrib]

lemma consecutive_potential_constant {n : ℕ} (a z : Fin (n+1) → ℝ)

  induction j using Fin.induction with
  | zero => rfl
  | succ i hi => have hh := h i; linarith




/-- Whole-matrix norm control inside any actual realization of the flattened
macro distances, so changing anchors does not require changing the embedding. -/
theorem whole_solve_in_flattened_realization {m : ℕ} (hm : 1 ≤ m)
    {K : ℝ} (hK : 10 ≤ K)
    (hlarge : 16384 * (harmonic (m+2) : ℝ)^2 ≤ K)
    (r : ℝ) (hr : r ≤ K^2/16)
    (p : Fin (m+2) → EuclideanSpace ℝ (Fin (m+1)))
    (hpd : ∀ i j, ‖p j-p i‖^2 = offDiagonalShift (colexSqData (m:=m+1) K) r i j)
    {σ : Type*} [Fintype σ] (Z : Matrix (Fin (m+1)) σ ℝ) :
    ∃ w : σ → EuclideanSpace ℝ (Fin (m+1)),
      (∀ i a, inner ℝ (p i.succ-p i.castSucc) (w a) = Z i a) ∧
      ∀ t : σ → ℝ, ‖∑ a, t a • w a‖^2 ≤
        144*(K+(m+1:ℕ)) * ∑ i, ((Z *ᵥ t) i / colexANat K (i.val+1))^2 := by
  let v : Fin (m+1) → EuclideanSpace ℝ (Fin (m+1)) :=
    fun i => p i.succ-p i.castSucc
  have hv : Matrix.gram ℝ v = incrementGram (offDiagonalShift (colexSqData (m:=m+1) K) r) :=
    gram_increments_of_sq_distances p _ hpd
  obtain ⟨X,hX,hbound⟩ := flattened_colex_whole_solve_energy hm hK hlarge r hr Z
  let w : σ → EuclideanSpace ℝ (Fin (m+1)) := fun a => ∑ j, X j a • v j
  refine ⟨w,?_,?_⟩
  · intro i a
    change inner ℝ (v i) (∑ j, X j a • v j) = _
    rw [gram_linear_combination_projection, hv]
    have hh := congrFun (congrFun hX i) a
    simpa only [Matrix.mul_apply, Matrix.mulVec, dotProduct] using hh
  · intro t
    have hw : ∑ a, t a • w a = ∑ j, (X *ᵥ t) j • v j := by
      simp only [w, Finset.smul_sum, smul_smul, Matrix.mulVec, dotProduct,
        Finset.sum_smul]
      rw [Finset.sum_comm]
      apply Finset.sum_congr rfl
      intro j hj
      apply Finset.sum_congr rfl
      intro a ha
      rw [mul_comm]
    rw [hw, gram_linear_combination_norm_sq, hv]
    exact hbound t

/-- A uniform denominator bound controls the entire fine matrix. -/
theorem fineTriangle_bound {n : ℕ} (P : Fin n → ℝ) {K : ℝ}
    (hK : 0 < K) (hn : (n:ℝ) ≤ K) (hP : ∀ i, K ≤ P i) (x : Fin n → ℝ) :
    (∑ i, (fineTriangle n P *ᵥ x) i ^ 2) ≤ ∑ i, x i ^ 2 := by
  have ha : ∀ i j, (fineTriangle n P i j)^2 ≤ 1/K^2 := by
    intro i j
    unfold fineTriangle
    split_ifs
    · rw [div_pow,one_pow]
      apply one_div_le_one_div_of_le (sq_pos_of_pos hK)
      nlinarith [hP i]
    · simp only [zero_pow (by norm_num : (2:ℕ) ≠ 0)]
      positivity
  have hx : 0 ≤ ∑ j, x j^2 := Finset.sum_nonneg (fun _ _ => sq_nonneg _)
  have hl : ∀ i, (fineTriangle n P *ᵥ x) i ^ 2 ≤ ((n:ℝ)/K^2)*∑ j, x j^2 := by
    intro i
    have hc := Finset.sum_mul_sq_le_sq_mul_sq Finset.univ (fineTriangle n P i) x
    have hs : ∑ j, (fineTriangle n P i j)^2 ≤ (n:ℝ)/K^2 := by
      calc
        _ ≤ ∑ j : Fin n, 1/K^2 := Finset.sum_le_sum fun j _ => ha i j
        _ = _ := by simp; ring
    exact hc.trans (mul_le_mul_of_nonneg_right hs hx)
  calc
    _ ≤ ∑ i : Fin n, ((n:ℝ)/K^2)*∑ j, x j^2 := Finset.sum_le_sum fun i _ => hl i
    _ = ((n:ℝ)^2/K^2)*∑ j, x j^2 := by simp; ring
    _ ≤ 1*∑ j, x j^2 := by
      apply mul_le_mul_of_nonneg_right _ hx
      apply (div_le_iff₀ (sq_pos_of_pos hK)).2
      nlinarith [sq_nonneg ((n:ℝ)-K)]
    _ = _ := one_mul _



/-- The coarse triangular estimate consumes the proved Hardy inequality,
not a per-column or unproved operator premise. -/
theorem coarseTriangle_bound {n : ℕ} (P : Fin n → ℝ)
    (hP : ∀ i, 0 < P i)
    (hquad : ∀ i, (i.val:ℝ)*(i.val+1) ≤ 2*P i) (x : Fin n → ℝ) :
    (∑ i, (coarseTriangle n P *ᵥ x) i ^ 2) ≤ 16*∑ i, x i ^ 2 := by
  let q : Fin n → ℝ := fun i => (i.val:ℝ)*(i.val+1)/P i
  have hq0 : ∀ i, 0 ≤ q i := by
    intro i
    exact div_nonneg (mul_nonneg (Nat.cast_nonneg _) (by positivity)) (hP i).le
  have hq2 : ∀ i, q i ≤ 2 := by
    intro i
    exact (div_le_iff₀ (hP i)).2 (hquad i)
  have he : ∀ i, (coarseTriangle n P *ᵥ x) i = q i*(hardyMatrix n *ᵥ x) i := by
    intro i
    simp only [Matrix.mulVec,dotProduct,Finset.mul_sum]
    apply Finset.sum_congr rfl
    intro j hj
    unfold coarseTriangle hardyMatrix q
    split_ifs
    · have hz : (i.val:ℝ)+1 ≠ 0 := by positivity
      norm_num only [Nat.cast_add,Nat.cast_one]
      field_simp
    · ring
  calc
    _ = ∑ i, (q i)^2*(hardyMatrix n *ᵥ x) i^2 := by simp_rw [he,mul_pow]
    _ ≤ ∑ i, 4*(hardyMatrix n *ᵥ x) i^2 := by
      apply Finset.sum_le_sum
      intro i hi
      apply mul_le_mul_of_nonneg_right _ (sq_nonneg _)
      nlinarith [hq0 i,hq2 i]
    _ = 4*∑ i, (hardyMatrix n *ᵥ x) i^2 := by rw [Finset.mul_sum]
    _ ≤ 4*(4*∑ i, x i^2) := mul_le_mul_of_nonneg_left (finite_hardy_two n x) (by norm_num)
    _ = _ := by ring


/-- The literal colex diagonal satisfies the coarse Hardy domination. -/

/-- Both complementary triangular bounds with the actual colex diagonal. -/
theorem colex_fine_coarse_triangle_bounds {n : ℕ} {K : ℝ}
    (hK : 0 < K) (hn : (n:ℝ) ≤ K) (x : Fin n → ℝ) :
    (∑ i, (fineTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i^2) ≤ ∑ i, x i^2 ∧
    (∑ i, (coarseTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i^2) ≤ 16*∑ i, x i^2 := by

    intro i
    unfold colexANat

  · apply coarseTriangle_bound
    · exact fun i => colex_triangle_positive hK i.val
    · exact fun i => colex_triangle_denominator hK.le i.val

/-- Adding the diagonal keeps the fine and coarse whole-matrix constants 2 and 5. -/
theorem colex_fine_coarse_full_bounds {n : ℕ} {K : ℝ}

    (∑ i, (x i+(coarseTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i)^2) ≤ 25*∑ i, x i^2 := by
  obtain ⟨hf,hc⟩ := colex_fine_coarse_triangle_bounds hK hn x
  constructor
  · calc
      _ ≤ ∑ i, (2*x i^2+2*(fineTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i^2) := by
        apply Finset.sum_le_sum
        intro i hi

      _ = 2*(∑ i, x i^2)+2*(∑ i, (fineTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i^2) := by
        rw [Finset.sum_add_distrib,← Finset.mul_sum,← Finset.mul_sum]
      _ ≤ _ := by linarith
  · calc
      _ ≤ ∑ i, (5*x i^2+(5/4:ℝ)*(coarseTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i^2) := by
        apply Finset.sum_le_sum
        intro i hi
        nlinarith [sq_nonneg (4*x i-(coarseTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i)]
      _ = 5*(∑ i, x i^2)+(5/4:ℝ)*(∑ i, (coarseTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i^2) := by
        rw [Finset.sum_add_distrib,← Finset.mul_sum,← Finset.mul_sum]
      _ ≤ _ := by linarith

/-- Literal fine potential differences give the diagonal minus strict upper ones. -/
theorem finePotential_incidence {n : ℕ} (K : ℝ) (i a : Fin n) :
    (pathIncidence n *ᵥ finePotential K a) i =
      (if i=a then colexANat K (i.val+1) else 0) -
      (if i.val < a.val then 1 else 0) := by
  rw [pathIncidence_mulVec_eq]
  simp only [finePotential, Fin.val_castSucc, Fin.val_succ]
  by_cases h : i=a
  · subst a
    simp [colexANat,Nat.choose_succ_succ]
    ring
  · have hv : i.val ≠ a.val := fun e => h (Fin.ext e)
    by_cases hia : i.val < a.val
    · have h₁ : i.val < a.val+1 := by omega
      have h₂ : i.val+1<a.val+1 := by omega
      simp only [h,h₁,h₂,hia,if_pos,if_false,Nat.cast_add,Nat.cast_one]
      ring
    · have h₁ : ¬ i.val < a.val+1 := by omega
      have h₂ : ¬ i.val+1<a.val+1 := by omega
      simp [h,h₁,h₂,hia]

/-- Literal coarse potential differences give minus diagonal and strict lower ramp. -/
theorem coarsePotential_incidence {n : ℕ} (K : ℝ) (i a : Fin n) :
    (pathIncidence n *ᵥ coarsePotential K a) i =
      -(if i=a then colexANat K (i.val+1) else 0) -
      (if a.val < i.val then (i.val:ℝ) else 0) := by
  rw [pathIncidence_mulVec_eq]
  simp only [coarsePotential, Fin.val_castSucc, Fin.val_succ]
  by_cases h : i=a
  · subst a
    simp [colexANat,Nat.choose_succ_succ]
    ring
  · have hv : i.val ≠ a.val := fun e => h (Fin.ext e)
    by_cases hai : a.val < i.val
    · have h₂ : a.val < i.val+1 := by omega
      have hc : Nat.choose (i.val+1) 2 = Nat.choose i.val 2+i.val := by
        simpa [Nat.add_comm] using Nat.choose_succ_succ i.val 1
      simp only [h,h₂,hai,if_pos,if_false,hc,Nat.cast_add,Nat.cast_one,neg_zero]
      ring
    · have h₂ : ¬ a.val < i.val+1 := by omega
      simp [h,h₂,hai]





lemma fineIncidence_normalized {n : ℕ} {K : ℝ} (hK : 0 < K) (x : Fin n → ℝ) (i : Fin n) :
    (fineIncidence n K *ᵥ x) i / colexANat K (i.val+1) =
      x i-(fineTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i := by
  have hp := ne_of_gt (colex_triangle_positive hK i.val)
  have hfi : fineIncidence n K = fun i a => (if i=a then colexANat K (i.val+1) else 0) - (if i.val < a.val then 1 else 0) := by
    ext i a
    exact finePotential_incidence K i a
  rw [hfi]
  simp only [Matrix.mulVec,dotProduct,
    sub_mul,ite_mul,zero_mul,Finset.sum_sub_distrib,sub_div,Finset.sum_div]
  simp only [ite_div,zero_div,Finset.sum_ite_eq,Finset.mem_univ,if_true]
  rw [mul_div_cancel_left₀ _ hp]
  congr 1
  apply Finset.sum_congr rfl
  intro a ha
  unfold fineTriangle
  split_ifs <;> simp <;> ring

lemma coarseIncidence_normalized {n : ℕ} {K : ℝ} (hK : 0 < K) (x : Fin n → ℝ) (i : Fin n) :
    (coarseIncidence n K *ᵥ x) i / colexANat K (i.val+1) =
      -(x i+(coarseTriangle n (fun j => colexANat K (j.val+1)) *ᵥ x) i) := by
  have hp := ne_of_gt (colex_triangle_positive hK i.val)
  have hci : coarseIncidence n K = fun i a => -(if i=a then colexANat K (i.val+1) else 0) - (if a.val < i.val then (i.val:ℝ) else 0) := by
    ext i a
    exact coarsePotential_incidence K i a
  rw [hci]
  simp only [Matrix.mulVec,dotProduct,
    sub_mul,neg_mul,ite_mul,zero_mul,Finset.sum_sub_distrib,Finset.sum_neg_distrib,
    sub_div,neg_div,Finset.sum_div]
  simp only [ite_div,zero_div,Finset.sum_ite_eq,Finset.mem_univ,if_true]
  rw [mul_div_cancel_left₀ _ hp]
  have hs : (∑ a : Fin n, (if a.val < i.val then (i.val:ℝ)*x a else 0)/colexANat K (i.val+1)) =
      ∑ a, coarseTriangle n (fun j => colexANat K (j.val+1)) i a*x a := by
    apply Finset.sum_congr rfl
    intro a ha
    unfold coarseTriangle
    split_ifs <;> simp <;> ring
  simp only [ite_div,zero_div] at hs
  rw [hs]
  ring

/-- Coordinates of a linear combination in the fixed macro Euclidean space. -/
theorem euclidean_sum_norm_sq_eq {ι κ : Type*} [Fintype ι] [Fintype κ]
    (B : κ → EuclideanSpace ℝ ι) (t : κ → ℝ) :
    ‖∑ a, t a • B a‖ ^ 2 =
      ∑ i : ι, (∑ a, t a * B a i) ^ 2 := by
  rw [EuclideanSpace.real_norm_sq_eq]
  congr 1
  ext i
  simp only [WithLp.ofLp_sum, WithLp.ofLp_smul, Finset.sum_apply,
    Pi.smul_apply, smul_eq_mul]

theorem sourceSlotMatrix_energy {ι κ : Type*}

/-- Inactive feature slots only add nonnegative diagonal reserve. -/
theorem source_three_active_slots_le_diagonal {κ : Type*}
    [Fintype κ] [DecidableEq κ]
    (b₀ b₁ : ℝ) (x : κ × Fin 5 → ℝ) :
    b₀ ^ 2 * (∑ a : κ, x (a,0) ^ 2) +
      b₁ ^ 2 * (∑ a : κ, x (a,1) ^ 2) +
      b₀ ^ 2 * (∑ a : κ, x (a,4) ^ 2) ≤
        x ⬝ᵥ (Matrix.diagonal
          (fun q : κ × Fin 5 => trigWeight b₀ b₁ b₀ q.2) *ᵥ x) := by
  have hpoint (a : κ) :
      b₀ ^ 2 * x (a,0) ^ 2 +
        b₁ ^ 2 * x (a,1) ^ 2 +
        b₀ ^ 2 * x (a,4) ^ 2 ≤
          ∑ s : Fin 5, trigWeight b₀ b₁ b₀ s * x (a,s) ^ 2 := by
    simp [Fin.sum_univ_succ, trigWeight]
    have h₂ : 0 ≤ b₁ ^ 2 * x (a,2) ^ 2 := by positivity
    have h₃ : 0 ≤ b₀ ^ 2 * x (a,3) ^ 2 := by positivity
    linarith
  rw [diagonal_bilinear_sum]
  rw [Fintype.sum_prod_type]
  have h := Finset.sum_le_sum (s := Finset.univ)
    (fun a (_ : a ∈ Finset.univ) => hpoint a)
  simpa only [Finset.sum_add_distrib, Finset.mul_sum, pow_two,
    mul_assoc] using h

/-- The actual three occupied feature slots, once their whole-family budgets
are known, admit a residual factor on the same finite feature index type. -/
theorem source_three_group_completion_factor {ι κ : Type*}
    [Fintype ι] [Fintype κ] [DecidableEq κ]
    (B U : κ → EuclideanSpace ℝ ι) (b₀ b₁ : ℝ)
    (hB : ∀ t : κ → ℝ,
      3 * ‖∑ a, t a • B a‖ ^ 2 ≤ b₀ ^ 2 * ∑ a, t a ^ 2)
    (hU : ∀ t : κ → ℝ,
      3 * ‖∑ a, t a • U a‖ ^ 2 ≤ b₁ ^ 2 * ∑ a, t a ^ 2) :
    ∃ V : Matrix (κ × Fin 5) (κ × Fin 5) ℝ,
      (sourceSlotMatrix B 0 + sourceSlotMatrix U 1 +
        sourceSlotMatrix (fun a => -B a) 4)ᵀ *
      (sourceSlotMatrix B 0 + sourceSlotMatrix U 1 +
        sourceSlotMatrix (fun a => -B a) 4) + Vᵀ * V =
        Matrix.diagonal (fun q : κ × Fin 5 => trigWeight b₀ b₁ b₀ q.2) := by
  let A₀ := sourceSlotMatrix B 0
  let A₁ := sourceSlotMatrix U 1
  let A₄ := sourceSlotMatrix (fun a => -B a) 4
  have hBcoord (t : κ → ℝ) :
      3 * (∑ i : ι, (∑ a : κ, t a * B a i) ^ 2) ≤
        b₀ ^ 2 * ∑ a, t a ^ 2 := by
    simpa only [euclidean_sum_norm_sq_eq] using hB t
  have hUcoord (t : κ → ℝ) :
      3 * (∑ i : ι, (∑ a : κ, t a * U a i) ^ 2) ≤
        b₁ ^ 2 * ∑ a, t a ^ 2 := by
    simpa only [euclidean_sum_norm_sq_eq] using hU t
  have hCnorm (t : κ → ℝ) :
      3 * ‖∑ a, t a • (-B a)‖ ^ 2 ≤
        b₀ ^ 2 * ∑ a, t a ^ 2 := by
    simpa only [smul_neg, Finset.sum_neg_distrib, norm_neg] using hB t
  have hCcoord (t : κ → ℝ) :
      3 * (∑ i : ι, (∑ a : κ, t a * (-B a) i) ^ 2) ≤
        b₀ ^ 2 * ∑ a, t a ^ 2 := by
    simpa only [euclidean_sum_norm_sq_eq] using hCnorm t
  have hbudget (x : κ × Fin 5 → ℝ) :
      3 * ∑ i, (A₀ *ᵥ x) i ^ 2 +
      3 * ∑ i, (A₁ *ᵥ x) i ^ 2 +
      3 * ∑ i, (A₄ *ᵥ x) i ^ 2 ≤
        x ⬝ᵥ (Matrix.diagonal
          (fun q : κ × Fin 5 => trigWeight b₀ b₁ b₀ q.2) *ᵥ x) := by
    have h₀ := sourceSlotMatrix_energy B 0 b₀ hBcoord x
    have h₁ := sourceSlotMatrix_energy U 1 b₁ hUcoord x
    have h₄ := sourceSlotMatrix_energy (fun a => -B a) 4 b₀ hCcoord x
    have hdiag := source_three_active_slots_le_diagonal b₀ b₁ x
    dsimp [A₀, A₁, A₄]
    linarith
  obtain ⟨V, hV⟩ := three_group_completion_factor A₀ A₁ A₄
    (Matrix.diagonal (fun q : κ × Fin 5 => trigWeight b₀ b₁ b₀ q.2))
    (Matrix.isHermitian_diagonal _) hbudget
  exact ⟨V, hV⟩

/-- The exact two-extraction operator estimate, stated as a quadratic form. -/
theorem sourceBReverseMatrix_sq_bound (n k : ℕ) (x : Fin (n+2) → ℝ) :
    (∑ i : Fin (n+1), (sourceBReverseMatrix n k *ᵥ x) i ^ 2) ≤
      ((64 / 9 : ℝ) + 9 * (k : ℝ)^2)^2 *
        ∑ A : Fin (n+2), x A ^ 2 := by
  let a : ℝ := 64 / 9
  let b : ℝ := 9 * (k : ℝ)^2
  have ha : 0 ≤ a := by dsimp [a]; norm_num
  have hb : 0 ≤ b := by dsimp [b]; positivity
  have hrow (i : Fin (n+1)) :
      (sourceBReverseMatrix n k *ᵥ x) i ^ 2 ≤
        (a+b) * (a * x i.succ ^ 2 + b * x i.castSucc ^ 2) := by
    rw [sourceBReverseMatrix_mulVec]
    have hid : (a+b) * (a * x i.succ ^ 2 + b * x i.castSucc ^ 2) -
        (-a*x i.succ+b*x i.castSucc)^2 =
          a*b*(x i.succ+x i.castSucc)^2 := by ring
    have hnonneg : 0 ≤ a*b*(x i.succ+x i.castSucc)^2 := by positivity
    dsimp [a,b] at hid
    nlinarith only [hid, hnonneg]
  have hup : (∑ i : Fin (n+1), x i.succ ^ 2) ≤
      ∑ A : Fin (n+2), x A ^ 2 := by
    simpa only [sourceBPickUpper_mulVec] using sourceBPickUpper_contract n x
  have hlo : (∑ i : Fin (n+1), x i.castSucc ^ 2) ≤
      ∑ A : Fin (n+2), x A ^ 2 := by
    simpa only [sourceBPickLower_mulVec] using sourceBPickLower_contract n x
  calc
    _ ≤ ∑ i : Fin (n+1),
        (a+b) * (a*x i.succ^2+b*x i.castSucc^2) :=
      Finset.sum_le_sum (fun i _ => hrow i)
    _ = (a+b) * (a*(∑ i : Fin (n+1), x i.succ^2) +
        b*(∑ i : Fin (n+1), x i.castSucc^2)) := by
      simp [Finset.mul_sum, Finset.sum_add_distrib, mul_add]
    _ ≤ (a+b) * (a*(∑ A : Fin (n+2), x A^2) +
        b*(∑ A : Fin (n+2), x A^2)) := by
      apply mul_le_mul_of_nonneg_left _ (add_nonneg ha hb)
      exact add_le_add (mul_le_mul_of_nonneg_left hup ha)
        (mul_le_mul_of_nonneg_left hlo hb)
    _ = _ := by dsimp [a,b]; ring

theorem sourceBReverseMatrix_weighted_sq_bound (n k : ℕ) {K : ℝ}
    (hK : 0 < K) (x : Fin (n+2) → ℝ) :
    (∑ i : Fin (n+1),
      ((sourceBReverseMatrix n k *ᵥ x) i /
        colexANat K (i.val+1))^2) ≤
      (((64 / 9 : ℝ) + 9*(k : ℝ)^2)/K)^2 *
        ∑ A : Fin (n+2), x A^2 := by
  have hweight (i : Fin (n+1)) : K ≤ colexANat K (i.val+1) := by
    unfold colexANat
    have hchoose : (0 : ℝ) ≤ Nat.choose (i.val+1+1) 2 := by positivity
    linarith
  have hterm (i : Fin (n+1)) :
      ((sourceBReverseMatrix n k *ᵥ x) i / colexANat K (i.val+1))^2 ≤
        ((sourceBReverseMatrix n k *ᵥ x) i / K)^2 := by
    have hp : 0 < colexANat K (i.val+1) := lt_of_lt_of_le hK (hweight i)
    have hs : K^2 ≤ (colexANat K (i.val+1))^2 := by nlinarith [hweight i]
    rw [div_pow, div_pow]
    apply (div_le_div_iff₀ (sq_pos_of_pos hp) (sq_pos_of_pos hK)).2
    exact mul_le_mul_of_nonneg_left hs (sq_nonneg _)
  calc
    _ ≤ ∑ i : Fin (n+1), ((sourceBReverseMatrix n k *ᵥ x) i / K)^2 :=
      Finset.sum_le_sum (fun i _ => hterm i)
    _ = (∑ i : Fin (n+1), (sourceBReverseMatrix n k *ᵥ x) i^2) / K^2 := by
      simp only [div_pow, Finset.sum_div]
    _ ≤ (((64 / 9 : ℝ) + 9*(k : ℝ)^2)^2 *
        ∑ A : Fin (n+2), x A^2) / K^2 := by
      gcongr
      exact sourceBReverseMatrix_sq_bound n k x
    _ = _ := by ring

theorem sourceBReverseMatrix_weighted_18_bound (n k : ℕ) (hk : 2 ≤ k)

/-- The matrix is the reverse difference of the actual rational anchor
potential, including the endpoint columns. -/
theorem sourceBReverseMatrix_eq_potential_difference (n k : ℕ)
    (i : Fin (n+1)) (A : Fin (n+2)) :
    sourceBReverseMatrix n k i A =
      sourceQuadraticPotential k A i.succ -
        sourceQuadraticPotential k A i.castSucc := by
  unfold sourceBReverseMatrix sourceBPickUpper sourceBPickLower
    sourceQuadraticPotential
  split_ifs <;> simp only [Fin.val_succ, Fin.val_castSucc, Fin.ext_iff] at *
    <;> first | omega | ring

/-- The existing whole solve, instantiated with the *actual* consecutive
quadratic-potential matrix. Its forward-edge input is `-Z_B`. -/
theorem sourceBQuadraticCoefficients_whole_solve {n k : ℕ} (hn : 1 ≤ n)
    (hk : 2 ≤ k) {K : ℝ} (hK : 10 ≤ K)
    (hlarge : 16384 * (harmonic (n+2) : ℝ)^2 ≤ K)
    (r : ℝ) (hr : r ≤ K^2/16)
    (p : Fin (n+2) → EuclideanSpace ℝ (Fin (n+1)))
    (hpd : ∀ i j, ‖p j-p i‖^2 =
      offDiagonalShift (colexSqData (m:=n+1) K) r i j) :
    ∃ B : Fin (n+2) → EuclideanSpace ℝ (Fin (n+1)),
      (∀ A J, inner ℝ (p A-p J) (B A) = sourceQuadraticPotential k A J) ∧
      (∀ t : Fin (n+2) → ℝ,
        ‖∑ A, t A • B A‖^2 ≤
          144*(K+(n+1:ℕ))*(18*(k : ℝ)^2/K)^2 *
            ∑ A, t A^2) := by
  let Z : Matrix (Fin (n+1)) (Fin (n+2)) ℝ :=
    fun i A => -sourceBReverseMatrix n k i A
  obtain ⟨B, hproj, henergy⟩ :=
    whole_solve_in_flattened_realization hn hK hlarge r hr p hpd Z
  refine ⟨B, ?_, ?_⟩
  · intro A J
    have hedges (i : Fin (n+1)) :
        inner ℝ (p i.succ) (B A) - inner ℝ (p i.castSucc) (B A) =
          sourceQuadraticPotential k A i.castSucc -
            sourceQuadraticPotential k A i.succ := by
      have h := hproj i A
      rw [inner_sub_left] at h
      change _ = -sourceBReverseMatrix n k i A at h
      rw [sourceBReverseMatrix_eq_potential_difference] at h
      linarith
    have ha := consecutive_potential_constant
      (fun j => inner ℝ (p j) (B A)) (sourceQuadraticPotential k A) hedges A
    have hj := consecutive_potential_constant
      (fun j => inner ℝ (p j) (B A)) (sourceQuadraticPotential k A) hedges J
    rw [sourceQuadraticPotential_self] at ha
    rw [inner_sub_left]
    linarith
  · intro t
    have hZ (i : Fin (n+1)) :
        (Z *ᵥ t) i = -(sourceBReverseMatrix n k *ᵥ t) i := by
      simp only [Z, Matrix.mulVec, dotProduct, neg_mul,
        Finset.sum_neg_distrib]
    have hweighted := sourceBReverseMatrix_weighted_18_bound n k hk
      (by linarith : 0 < K) t
    calc
      ‖∑ A, t A • B A‖^2 ≤
          144*(K+(n+1:ℕ)) *
            ∑ i, ((Z *ᵥ t) i / colexANat K (i.val+1))^2 := henergy t
      _ = 144*(K+(n+1:ℕ)) *
            ∑ i, ((sourceBReverseMatrix n k *ᵥ t) i /
              colexANat K (i.val+1))^2 := by simp only [hZ, neg_div, even_two, Even.neg_pow]
      _ ≤ 144*(K+(n+1:ℕ)) *
            ((18*(k : ℝ)^2/K)^2 * ∑ A, t A^2) :=
        mul_le_mul_of_nonneg_left hweighted (by positivity)
      _ = _ := by ring

/-- The whole-solve B estimate fits *each* of the two equal quadratic
completion budgets, using the actual ceiling offset and source range. -/
theorem sourceBQuadraticScalar_budget {n k : ℕ} (hm : 34000 ≤ n+2) :
    3*(144*((sourceOffset (n+2) : ℝ)+(n+1:ℕ)) *
      (18*(k : ℝ)^2/(sourceOffset (n+2) : ℝ))^2) ≤
      sourceReserve0Sq (n+2) k := by
  have hupper := sourceOffset_plus_m_upper (n+2) hm
  have hcount : (((n+1 : ℕ) : ℝ) ≤ (n+2:ℕ)) := by
    exact_mod_cast (show n+1 ≤ n+2 by omega)
  have hmid : (sourceOffset (n+2) : ℝ)+(n+1:ℕ) ≤
      ((n+2:ℕ) : ℝ)^2/63 := by linarith
  have hfactor : 144*((sourceOffset (n+2) : ℝ)+(n+1:ℕ)) ≤
      4*((n+2:ℕ) : ℝ)^2 := by
    have hscaled := mul_le_mul_of_nonneg_left hmid
      (by norm_num : (0 : ℝ) ≤ 144)
    nlinarith only [hscaled, sq_nonneg (((n+2:ℕ) : ℝ))]
  have hfactor' := mul_le_mul_of_nonneg_right hfactor
    (sq_nonneg (18*(k : ℝ)^2/(sourceOffset (n+2) : ℝ)))
  calc
    _ = 3*((144*((sourceOffset (n+2) : ℝ)+(n+1:ℕ))) *
        (18*(k : ℝ)^2/(sourceOffset (n+2) : ℝ))^2) := by ring
    _ ≤ 3*((4*((n+2:ℕ) : ℝ)^2) *
        (18*(k : ℝ)^2/(sourceOffset (n+2) : ℝ))^2) :=
      mul_le_mul_of_nonneg_left hfactor' (by norm_num)
    _ = sourceReserve0Sq (n+2) k := by
      unfold sourceReserve0Sq
      ring

/-- Same-index vectors for the actual flattened source centers, with their
quadratic projections and a whole-family bound in the paper's `b₀²/3 = b₂²/3`
completion budget. -/
theorem sourceBQuadraticCoefficients_source_budget {n k : ℕ}
    (hm : 34000 ≤ n+2) (hk : 2 ≤ k)
    (p : Fin (n+2) → EuclideanSpace ℝ (Fin (n+1)))
    (hpd : ∀ i j, ‖p j-p i‖^2 =
      offDiagonalShift (colexSqData (m:=n+1)
        (sourceOffset (n+2) : ℝ))
        (sourceFlatteningRadius (n+2) k) i j) :
    ∃ B : Fin (n+2) → EuclideanSpace ℝ (Fin (n+1)),
      (∀ A J, inner ℝ (p A-p J) (B A) = sourceQuadraticPotential k A J) ∧
      (∀ t : Fin (n+2) → ℝ,
        3*‖∑ A, t A • B A‖^2 ≤
          sourceReserve0Sq (n+2) k * ∑ A, t A^2) := by
  obtain ⟨hK, hlarge, _, hr⟩ :=
    source_macro_operator_scalar_gates (n+2) k hm hk
  obtain ⟨B, hproj, hbound⟩ :=
    sourceBQuadraticCoefficients_whole_solve (by omega : 1 ≤ n)
      hk hK hlarge (sourceFlatteningRadius (n+2) k) hr p hpd
  refine ⟨B, hproj, ?_⟩
  intro t
  have hs := sourceBQuadraticScalar_budget (k:=k) hm
  have hsum : 0 ≤ ∑ A : Fin (n+2), t A^2 :=
    Finset.sum_nonneg (fun A _ => sq_nonneg _)
  calc
    3*‖∑ A, t A • B A‖^2 ≤
        3*((144*((sourceOffset (n+2) : ℝ)+(n+1:ℕ)) *
          (18*(k : ℝ)^2/(sourceOffset (n+2) : ℝ))^2) *
            ∑ A, t A^2) :=
      mul_le_mul_of_nonneg_left (hbound t) (by norm_num)
    _ = (3*(144*((sourceOffset (n+2) : ℝ)+(n+1:ℕ)) *
          (18*(k : ℝ)^2/(sourceOffset (n+2) : ℝ))^2)) *
            ∑ A, t A^2 := by ring
    _ ≤ sourceReserve0Sq (n+2) k * ∑ A, t A^2 :=
      mul_le_mul_of_nonneg_right hs hsum







/-- Acyclic downstream assembly of the literal source operator. This uses
the two-diagonal *whole-matrix* quadratic B estimate when constructing the
residual Gram factor; the older per-column assembly remains available. -/
theorem source_actual_operator_completion_universal_B_matrix {n k : ℕ}
    (hm : 34000 ≤ n+2) (hk : 2 ≤ k) :
    ∃ p : Fin (n+2) → EuclideanSpace ℝ (Fin (n+1)),
    ∃ B : Fin (n+2) → EuclideanSpace ℝ (Fin (n+1)),
    ∃ F G : Fin (n+1) → EuclideanSpace ℝ (Fin (n+1)),
    ∃ V : Matrix (Fin (n+2) × Fin 5) (Fin (n+2) × Fin 5) ℝ,
      (∀ i j, ‖p j-p i‖ ^ 2 =
        offDiagonalShift (colexSqData (m:=n+1)
          (sourceOffset (n+2) : ℝ))
          (sourceFlatteningRadius (n+2) k) i j) ∧
      (∀ A J, inner ℝ (p A-p J) (B A) =
        sourceQuadraticPotential k A J) ∧
      (∀ i a, inner ℝ (p i.succ-p i.castSucc) (F a) =
        fineIncidence (n+1) (sourceOffset (n+2) : ℝ) i a) ∧
      (∀ i a, inner ℝ (p i.succ-p i.castSucc) (G a) =
        coarseIncidence (n+1) (sourceOffset (n+2) : ℝ) i a) ∧
      (sourceActiveMatrix k B F G)ᵀ * (sourceActiveMatrix k B F G) +
        Vᵀ * V =
        Matrix.diagonal (fun q : Fin (n+2) × Fin 5 =>
          trigWeight (sourceReserve0 (n+2) k)
            (sourceReserve1 (n+2) k)
            (sourceReserve0 (n+2) k) q.2) := by
  obtain ⟨hK, hlarge, hmacro, hr⟩ :=
    source_macro_operator_scalar_gates (n+2) k hm hk
  have hnK : (n+1 : ℕ) ≤ (sourceOffset (n+2) : ℝ) := by
    have hh : ((n+1 : ℕ) : ℝ) ≤ (n+2 : ℕ) := by
      exact_mod_cast (show n+1 ≤ n+2 by omega)
    linarith
  obtain ⟨p, _, hpd⟩ :=
    flattened_colex_realizes (by omega : 1 ≤ n) hK hlarge
      (sourceFlatteningRadius (n+2) k) hr
  obtain ⟨B, hBproj, hBbudget⟩ :=
    sourceBQuadraticCoefficients_source_budget hm hk p hpd
  obtain ⟨F, G, hFproj, hGproj, hFbudget, hGbudget⟩ :=
    sourceFineCoarseCoefficients_exist (by omega : 1 ≤ n) hK hlarge
      (sourceFlatteningRadius (n+2) k) hr hnK p hpd
  have hBnorm (t : Fin (n+2) → ℝ) :
      3 * ‖∑ A, t A • B A‖ ^ 2 ≤
        (sourceReserve0 (n+2) k) ^ 2 * ∑ A, t A ^ 2 := by
    rw [sourceReserve0_square]
    exact hBbudget t
  have hUnorm (t : Fin (n+2) → ℝ) :
      3 * ‖∑ A, t A • sourceSineCoefficient k F G A‖ ^ 2 ≤
        (sourceReserve1 (n+2) k) ^ 2 * ∑ A, t A ^ 2 := by
    rw [sourceReserve1_square]
    exact sourceSineCoefficients_whole_budget k hm hk F G
      hFbudget hGbudget t
  obtain ⟨V, hV⟩ := source_three_group_completion_factor B
    (sourceSineCoefficient k F G)
    (sourceReserve0 (n+2) k) (sourceReserve1 (n+2) k)
    hBnorm hUnorm
  exact ⟨p, B, F, G, V, hpd, hBproj, hFproj, hGproj, hV⟩

end Erdos488.Geometry.SeparatedDistances.FixedDimension

theorem _root_.solution {n k : ℕ}
    (hm : 34000 ≤ n+2) (hk : 2 ≤ k) (prime : ℕ) :
    ∃ P : Fin (n+2) × Fin k →
        EuclideanSpace ℝ (ClusterCoordinate (Fin (n+1)) (n+2)),
      (∀ A : Fin (n+2), ∀ i j : Fin k, i.val < j.val →
        dist (P (A,j)) (P (A,i)) =
          sourceLocalDistance prime (n+2) k
            (sourceReserve1 (n+2) k) (sourceReserve0 (n+2) k)
            A.val i.val (j.val-i.val)) ∧
      (∀ A J : Fin (n+2), A.val < J.val →
        ∀ i j : Fin k,
          dist (P (A,i)) (P (J,j))^2 =
            (sourceCrossBaseline (sourceOffset (n+2) : ℝ) k
                (sourcePairCode A J) +
              sourceAlpha k i.val + sourceBeta k j.val)^2 -
              2*sourceAlpha k i.val*sourceBeta k j.val) := by
  obtain ⟨p, B, F, G, V, hpd, hB, hF, hG, hfactor⟩ :=
    source_actual_operator_completion_universal_B_matrix hm hk
  let R := sourceRadius (n+2) k
  let b₀ := sourceReserve0 (n+2) k
  let b₁ := sourceReserve1 (n+2) k
  let axis := Real.sqrt (R^2-(2*b₀^2+b₁^2))
  let P : Fin (n+2) × Fin k →
      EuclideanSpace ℝ (ClusterCoordinate (Fin (n+1)) (n+2)) :=
    fun q => rulerPoint (fun A t => p A t)
      (sourceActiveMatrix k B F G) V
      (sourceMacroStep (k : ℝ)) R axis (sourceTheta k)
      (((k : ℝ)-1)/2) prime k q.1 q.2.val
  refine ⟨P, ?_, ?_⟩
  · intro A i j hij
    have hR : R ≠ 0 :=
      (sourceRadius_pos (n+2) k (by omega) (by omega)).ne'
    have hdist := rulerPoint_local_distance (fun A t => p A t)
      (sourceActiveMatrix k B F G) V
      (sourceMacroStep (k : ℝ)) R axis (sourceTheta k)
      (((k : ℝ)-1)/2) b₀ b₁ b₀ prime k A i.val
      (j.val-i.val) hR hfactor
    have hsum : i.val+(j.val-i.val)=j.val := by omega
    simpa only [P, hsum, sourceLocalDistance, R, b₀, b₁] using hdist
  · intro A J hAJ i j
    have hreserve : 2*b₀^2+b₁^2 ≤ R^2 := by
      dsimp [b₀, b₁, R]
      rw [sourceReserve0_square, sourceReserve1_square]
      exact source_total_reserve_bound (n+2) k hm
    change dist
      (clusterPoint (fun A t => p A t) (sourceActiveMatrix k B F G) V
        (sourceMacroStep (k : ℝ)) R axis A
        (sourceTheta k * ((i.val : ℝ)-((k : ℝ)-1)/2))
        (3*(primeRuler prime k A.val i.val : ℝ)))
      (clusterPoint (fun A t => p A t) (sourceActiveMatrix k B F G) V
        (sourceMacroStep (k : ℝ)) R axis J
        (sourceTheta k * ((j.val : ℝ)-((k : ℝ)-1)/2))
        (3*(primeRuler prime k J.val j.val : ℝ))) ^ 2 = _
    exact source_clusterPoint_cross_metric_square (by omega) _ p B F G V
      hpd hB hF hG b₀ b₁ hfactor hreserve A J hAJ
      i.val j.val
      (3*(primeRuler prime k A.val i.val : ℝ))
      (3*(primeRuler prime k J.val j.val : ℝ))

end Erdos488.Geometry.SeparatedDistances.FixedDimension
