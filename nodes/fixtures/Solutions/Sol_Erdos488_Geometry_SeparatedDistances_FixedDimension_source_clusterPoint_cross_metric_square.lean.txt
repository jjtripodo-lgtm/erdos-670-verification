/-- Pure-colex prescribed distances are symmetric. -/
theorem colexDistance_symm {m : ℕ} (K : ℝ) (i j : Fin (m + 1)) :
    colexDistance K i j = colexDistance K j i := by
  by_cases hij : i.val < j.val
  · simp [colexDistance, hij, Nat.not_lt.mpr hij.le]
  · by_cases hji : j.val < i.val
    · simp [colexDistance, hji, Nat.not_lt.mpr hji.le]
    · have h : i = j := Fin.ext (Nat.le_antisymm (Nat.not_lt.mp hji) (Nat.not_lt.mp hij))
      subst j
      simp

/-- Exact cancellation of the combined macro and residual inner products. -/
theorem completed_bilinear_identity
    {ι κ : Type*} [Fintype ι] [Fintype κ]
    (M : Matrix ι κ ℝ) (V D : Matrix κ κ ℝ)
    (hfactor : Mᵀ*M + Vᵀ*V = D) (x y : κ → ℝ) :
    (M *ᵥ x) ⬝ᵥ (M *ᵥ y) + (V *ᵥ x) ⬝ᵥ (V *ᵥ y) =
      x ⬝ᵥ (D *ᵥ y) := by
  rw [← hfactor, Matrix.add_mulVec, dotProduct_add]
  congr 1 <;> rw [← Matrix.mulVec_mulVec] <;> symm <;>
    rw [Matrix.dotProduct_mulVec, Matrix.vecMul_transpose]

/-- The weighted feature norm is independent of the local point. -/
theorem trigFeature_constant_norm (b₀ b₁ b₂ z : ℝ) :
    (∑ a : Fin 5, trigWeight b₀ b₁ b₂ a*(trigFeature z a)^2)=b₀^2+b₁^2+b₂^2 := by
  simp only [Fin.sum_univ_succ,trigFeature,trigWeight,Matrix.cons_val_zero,Matrix.cons_val_succ,
    Matrix.cons_val_fin_one,Fin.sum_univ_zero,add_zero,one_pow,mul_one]
  have h₁ := congrArg (fun x : ℝ => b₁^2*x) (Real.sin_sq_add_cos_sq z)
  have h₂ := congrArg (fun x : ℝ => b₂^2*x) (Real.sin_sq_add_cos_sq (2*z))
  nlinarith only [h₁,h₂]









/-- Literal block disjointness, with no orthogonality assumption. -/
theorem clusteredFeature_weighted_dot {m : ℕ} (A B : Fin m) (b₀ b₁ b₂ z w : ℝ) :
    (∑ q : Fin m × Fin 5, trigWeight b₀ b₁ b₂ q.2*
      clusteredFeature A z q*clusteredFeature B w q)=
    if A=B then ∑ a : Fin 5, trigWeight b₀ b₁ b₂ a*trigFeature z a*trigFeature w a else 0 := by
  rw [Fintype.sum_prod_type]
  have he : ∀ C : Fin m,
      (∑ a : Fin 5, trigWeight b₀ b₁ b₂ a*clusteredFeature A z (C,a)*
        clusteredFeature B w (C,a)) =
      if C=A then (if A=B then ∑ a : Fin 5,
        trigWeight b₀ b₁ b₂ a*trigFeature z a*trigFeature w a else 0) else 0 := by
    intro C
    by_cases hC : C=A
    · subst C
      by_cases hAB : A=B <;> simp [clusteredFeature,hAB]
    · simp [clusteredFeature,hC]
  simp_rw [he]
  simp

/-- Constant feature reserve survives embedding in the full 5m-coordinate space. -/
theorem clusteredFeature_constant_norm {m : ℕ} (A : Fin m) (b₀ b₁ b₂ z : ℝ) :
    (∑ q : Fin m × Fin 5, trigWeight b₀ b₁ b₂ q.2*(clusteredFeature A z q)^2)=
      b₀^2+b₁^2+b₂^2 := by
  have h := clusteredFeature_weighted_dot A A b₀ b₁ b₂ z z
  simp only [if_pos rfl] at h
  simp_rw [mul_assoc,← sq] at h
  exact h.trans (trigFeature_constant_norm b₀ b₁ b₂ z)

theorem diagonal_bilinear_sum {κ : Type*} [Fintype κ] [DecidableEq κ]
    (w x y : κ → ℝ) :
    x ⬝ᵥ ((Matrix.diagonal w) *ᵥ y) = ∑ q, w q*x q*y q := by
  simp only [Matrix.mulVec_diagonal,dotProduct,Pi.mul_apply]
  apply Finset.sum_congr rfl
  intro q _
  ring

/-- Off-cluster cancellation for the actual embedded five-coordinate features. -/
theorem completed_feature_cross_cancel {ι : Type*} [Fintype ι] {m : ℕ}
    (M : Matrix ι (Fin m × Fin 5) ℝ) (V : Matrix (Fin m × Fin 5) (Fin m × Fin 5) ℝ)
    (b₀ b₁ b₂ : ℝ)
    (hfactor : Mᵀ*M+Vᵀ*V=Matrix.diagonal (fun q => trigWeight b₀ b₁ b₂ q.2))
    (A B : Fin m) (hAB : A ≠ B) (z w : ℝ) :
    (M *ᵥ clusteredFeature A z) ⬝ᵥ (M *ᵥ clusteredFeature B w)+
      (V *ᵥ clusteredFeature A z) ⬝ᵥ (V *ᵥ clusteredFeature B w)=0 := by
  rw [completed_bilinear_identity M V _ hfactor,diagonal_bilinear_sum,
    clusteredFeature_weighted_dot,if_neg hAB]

/-- The completed macro-plus-residual perturbation has the exact common norm. -/
theorem completed_feature_constant_norm {ι : Type*} [Fintype ι] {m : ℕ}
    (M : Matrix ι (Fin m × Fin 5) ℝ) (V : Matrix (Fin m × Fin 5) (Fin m × Fin 5) ℝ)
    (b₀ b₁ b₂ : ℝ)
    (hfactor : Mᵀ*M+Vᵀ*V=Matrix.diagonal (fun q => trigWeight b₀ b₁ b₂ q.2))
    (A : Fin m) (z : ℝ) :
    (∑ q, ((M *ᵥ clusteredFeature A z) q)^2)+
      (∑ q, ((V *ᵥ clusteredFeature A z) q)^2)=b₀^2+b₁^2+b₂^2 := by
  have h := completed_bilinear_identity M V _ hfactor (clusteredFeature A z) (clusteredFeature A z)
  rw [diagonal_bilinear_sum] at h
  simp_rw [dotProduct,mul_assoc,← sq] at h
  exact h.trans (clusteredFeature_constant_norm A b₀ b₁ b₂ z)

/-- Exact finite sum identity used to expand Euclidean squared distances. -/
theorem sum_sq_sub {κ : Type*} [Fintype κ] (x y : κ → ℝ) :
    (∑ q, (x q-y q)^2)=(∑ q, (x q)^2)+(∑ q, (y q)^2)-2*(x ⬝ᵥ y) := by
  have he : ∀ q, (x q-y q)^2=(x q)^2+(y q)^2-2*(x q*y q) := by
    intro q
    ring
  simp_rw [he]
  simp only [Finset.sum_add_distrib,Finset.sum_sub_distrib,dotProduct,← Finset.mul_sum]

/-- Squared feature distance across distinct clusters is twice the reserve. -/
theorem completed_cross_feature_square {ι : Type*} [Fintype ι] {m : ℕ}
    (M : Matrix ι (Fin m × Fin 5) ℝ) (V : Matrix (Fin m × Fin 5) (Fin m × Fin 5) ℝ)
    (b₀ b₁ b₂ : ℝ)
    (hfactor : Mᵀ*M+Vᵀ*V=Matrix.diagonal (fun q => trigWeight b₀ b₁ b₂ q.2))
    (A B : Fin m) (hAB : A ≠ B) (z w : ℝ) :
    (∑ q, ((M *ᵥ clusteredFeature A z) q-(M *ᵥ clusteredFeature B w) q)^2)+
      (∑ q, ((V *ᵥ clusteredFeature A z) q-(V *ᵥ clusteredFeature B w) q)^2)=
        2*(b₀^2+b₁^2+b₂^2) := by
  rw [sum_sq_sub,sum_sq_sub]
  have h₀ := completed_feature_constant_norm M V b₀ b₁ b₂ hfactor A z
  have h₁ := completed_feature_constant_norm M V b₀ b₁ b₂ hfactor B w
  have hcross := completed_feature_cross_cancel M V b₀ b₁ b₂ hfactor A B hAB z w
  linarith

theorem privateCircle_norm_square {m : ℕ} (A : Fin m) (R y : ℝ) :
    (∑ q, (privateCircle A R y q)^2)=R^2 := by
  rw [Fintype.sum_prod_type]
  have he : ∀ B : Fin m, (∑ a : Fin 2, (privateCircle A R y (B,a))^2)=
      if B=A then (R*Real.cos (y/R))^2+(R*Real.sin (y/R))^2 else 0 := by
    intro B
    by_cases h : B=A <;> simp [privateCircle,h,Fin.sum_univ_succ]
  simp_rw [he]
  simp only [Finset.sum_ite_eq',Finset.mem_univ,if_true]
  have h := congrArg (fun x : ℝ => R^2*x) (Real.sin_sq_add_cos_sq (y/R))
  nlinarith only [h]

theorem privateCircle_cross_dot {m : ℕ} (A B : Fin m) (hAB : A ≠ B) (R yj yi : ℝ) :
    privateCircle A R yj ⬝ᵥ privateCircle B R yi=0 := by
  apply Finset.sum_eq_zero
  intro q _
  by_cases h : q.1=A
  · have h' : q.1 ≠ B := by simpa [h] using hAB
    simp [privateCircle,h,hAB]
  · simp [privateCircle,h]

theorem privateCircle_cross_square {m : ℕ} (A B : Fin m) (hAB : A ≠ B) (R yj yi : ℝ) :
    (∑ q, (privateCircle A R yj q-privateCircle B R yi q)^2)=2*R^2 := by
  rw [sum_sq_sub,privateCircle_norm_square,privateCircle_norm_square,
    privateCircle_cross_dot A B hAB]
  ring

theorem privateAxis_cross_square {m : ℕ} (A B : Fin m) (hAB : A ≠ B) (axis : ℝ) :
    (∑ C : Fin m, ((if C=A then axis else 0)-(if C=B then axis else 0))^2)=2*axis^2 := by
  have he : ∀ C : Fin m, ((if C=A then axis else 0)-(if C=B then axis else 0))^2=
      (if C=A then axis^2 else 0)+(if C=B then axis^2 else 0) := by
    intro C
    by_cases ha : C=A
    · have hb : C ≠ B := by simpa [ha] using hAB
      simp [ha,hb,hAB,Ne.symm hAB]
    · by_cases hb : C=B <;> simp [ha,hb,hAB,Ne.symm hAB]
  simp_rw [he]
  simp [Finset.sum_add_distrib]
  ring

/-- General macro-coordinate expansion, with both signed projection terms retained. -/
theorem macro_coordinate_expansion {ι : Type*} [Fintype ι] (a b u v : ι → ℝ) (S : ℝ) :
    (∑ q, (S*a q+u q-(S*b q+v q))^2)=
      S^2*(∑ q, (a q-b q)^2)+2*S*((a-b) ⬝ᵥ (u-v))+(∑ q, (u q-v q)^2) := by
  have he : ∀ q, (S*a q+u q-(S*b q+v q))^2=
      S^2*(a q-b q)^2+2*S*((a q-b q)*(u q-v q))+(u q-v q)^2 := by
    intro q
    ring
  simp_rw [he]
  simp only [Finset.sum_add_distrib,← Finset.mul_sum,dotProduct,Pi.sub_apply]

/-- The actual cross-cluster Euclidean square, before imposing projection targets. -/
theorem clusterPoint_cross_square {ι : Type*} [Fintype ι] {m : ℕ}
    (centers : Fin m → ι → ℝ) (M : Matrix ι (Fin m × Fin 5) ℝ)
    (V : Matrix (Fin m × Fin 5) (Fin m × Fin 5) ℝ) (S R axis b₀ b₁ b₂ : ℝ)
    (hfactor : Mᵀ*M+Vᵀ*V=Matrix.diagonal (fun q => trigWeight b₀ b₁ b₂ q.2))
    (A B : Fin m) (hAB : A ≠ B) (z w yj yi : ℝ) :
    dist (clusterPoint centers M V S R axis A z yj)
      (clusterPoint centers M V S R axis B w yi)^2=
      S^2*(∑ q, (centers A q-centers B q)^2)+
      2*S*((centers A-centers B) ⬝ᵥ
        ((M *ᵥ clusteredFeature A z)-(M *ᵥ clusteredFeature B w)))+
      2*(R^2+axis^2+b₀^2+b₁^2+b₂^2) := by
  rw [EuclideanSpace.dist_sq_eq]
  simp only [clusterPoint,WithLp.ofLp_toLp,Fintype.sum_sum_type,Sum.elim_inl,Sum.elim_inr,
    Real.dist_eq,sq_abs]
  rw [macro_coordinate_expansion,privateCircle_cross_square A B hAB,
    privateAxis_cross_square A B hAB]
  have h := completed_cross_feature_square M V b₀ b₁ b₂ hfactor A B hAB z w
  linarith

/-- The exact cross metric now follows from actual Euclidean coordinates and
both directed projection equations; the mixed product is retained. -/
theorem clusterPoint_cross_metric_square {ι : Type*} [Fintype ι] {m : ℕ}
    (centers : Fin m → ι → ℝ) (M : Matrix ι (Fin m × Fin 5) ℝ)
    (V : Matrix (Fin m × Fin 5) (Fin m × Fin 5) ℝ) (S R axis b₀ b₁ b₂ ν α β : ℝ)
    (hfactor : Mᵀ*M+Vᵀ*V=Matrix.diagonal (fun q => trigWeight b₀ b₁ b₂ q.2))
    (A B : Fin m) (hAB : A ≠ B) (z w yj yi : ℝ)
    (haxis : axis^2=R^2-(b₀^2+b₁^2+b₂^2))
    (hbase : S^2*(∑ q, (centers A q-centers B q)^2)+4*R^2=ν^2)
    (hprojA : 2*S*((centers A-centers B) ⬝ᵥ (M *ᵥ clusteredFeature A z))=2*ν*α+α^2)

    dist (clusterPoint centers M V S R axis A z yj)

  have h := clusterPoint_cross_square centers M V S R axis b₀ b₁ b₂ hfactor A B hAB z w yj yi
  rw [dotProduct_sub] at h
  nlinarith

/-- Incidence projection equations telescope with the sign convention of the
actual macro increments. -/
theorem anchored_projection_of_pathIncidence
    {n : ℕ} (p : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (w : EuclideanSpace ℝ (Fin n)) (z : Fin (n+1) → ℝ)
    (A : Fin (n+1)) (hA : z A = 0)
    (hedge : ∀ i : Fin n,
      inner ℝ (p i.succ-p i.castSucc) w =
        (pathIncidence n *ᵥ z) i) (J : Fin (n+1)) :
    inner ℝ (p A-p J) w = z J := by
  have hedges : ∀ i : Fin n,
      inner ℝ (p i.succ) w-inner ℝ (p i.castSucc) w =
        z i.castSucc-z i.succ := by
    intro i
    simpa only [inner_sub_left, pathIncidence_mulVec_eq] using hedge i
  have ha := consecutive_potential_constant
    (fun V => inner ℝ (p V) w) z hedges A
  have hj := consecutive_potential_constant
    (fun V => inner ℝ (p V) w) z hedges J
  rw [hA] at ha
  rw [inner_sub_left]
  linarith

vertex; its anchor is `a+1`. -/

    (hF : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (F a) =
      fineIncidence n K i a) (A J : Fin (n+1)) :
    inner ℝ (p A-p J) (sourceFineShift F A) =
      if J.val < A.val then colexDistance K A J else 0 := by
  cases A using Fin.cases with
  | zero => simp [sourceFineShift]
  | succ a =>
      simpa only [sourceFineShift, Fin.cases_succ] using
        (sourceFineCoefficient_anchor_projection K p F hF a J).trans
          (finePotential_eq_colex K a J)

theorem sourceCoarseShift_anchor_projection {n : ℕ} (K : ℝ)
    (p : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (G : Fin n → EuclideanSpace ℝ (Fin n))
    (hG : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (G a) =
      coarseIncidence n K i a) (A J : Fin (n+1)) :
    inner ℝ (p A-p J) (sourceCoarseShift G A) =
      if A.val < J.val then colexDistance K A J else 0 := by
  cases A using Fin.lastCases with
  | last =>
      have h : ¬ n < J.val := by omega
      simp [sourceCoarseShift, h]
  | cast a =>
      simpa only [sourceCoarseShift, Fin.snoc_castSucc] using
        (sourceCoarseCoefficient_anchor_projection K p G hG a J).trans
          (coarsePotential_eq_colex K a J)





















theorem sourceSineCoefficient_anchor_projection {n : ℕ} (K : ℝ) (k : ℕ)
    (p : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (F G : Fin n → EuclideanSpace ℝ (Fin n))
    (hF : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (F a) =
      fineIncidence n K i a)
    (hG : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (G a) =
      coarseIncidence n K i a)
    (A J : Fin (n+1)) (hAJ : A ≠ J) :
    inner ℝ (p A-p J) (sourceSineCoefficient k F G A) =
      ((9/8 : ℝ) / sourceTheta k) *
        sourceDirectedScale k A J * colexDistance K A J := by
  rw [sourceSineCoefficient, real_inner_smul_right, inner_add_right,
    real_inner_smul_right]
  rw [sourceFineShift_anchor_projection K p F hF A J,
    sourceCoarseShift_anchor_projection K p G hG A J]
  by_cases hlt : A.val < J.val
  · have hnot : ¬J.val < A.val := by omega
    simp [sourceDirectedScale, hlt, hnot]
    ring
  · have hgt : J.val < A.val := by
      have hne : A.val ≠ J.val := fun h => hAJ (Fin.ext h)
      omega
    simp [sourceDirectedScale, hlt, hgt]

theorem euclidean_dot_inner {ι : Type*} [Fintype ι]
    (v w : EuclideanSpace ℝ ι) :

/-- Literal source feature projection in one actual macro realization.
The rational potential supplies the second-harmonic term and the shifted
incidence operator supplies the first-harmonic term. -/
theorem sourceActiveMatrix_anchor_projection {n : ℕ} (K : ℝ) (k : ℕ)
    (p : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (B : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (F G : Fin n → EuclideanSpace ℝ (Fin n))
    (hB : ∀ A J, inner ℝ (p A-p J) (B A) =
      sourceQuadraticPotential k A J)
    (hF : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (F a) =
      fineIncidence n K i a)
    (hG : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (G a) =
      coarseIncidence n K i a)
    (A J : Fin (n+1)) (hAJ : A ≠ J) (z : ℝ) :
    (fun i => (p A-p J) i) ⬝ᵥ
      (sourceActiveMatrix k B F G *ᵥ clusteredFeature A z) =
      sourceQuadraticPotential k A J * (1-Real.cos (2*z)) +
        ((9/8 : ℝ) / sourceTheta k) *
          sourceDirectedScale k A J * colexDistance K A J * Real.sin z := by
  have hdotB : (∑ i : Fin n, (p A-p J) i * B A i) =
      inner ℝ (p A-p J) (B A) := euclidean_dot_inner _ _
  have hdotU : (∑ i : Fin n,
      (p A-p J) i * sourceSineCoefficient k F G A i) =
      inner ℝ (p A-p J) (sourceSineCoefficient k F G A) :=
    euclidean_dot_inner _ _
  calc
    _ = (∑ i : Fin n, (p A-p J) i * B A i) *
          (1-Real.cos (2*z)) +
        (∑ i : Fin n,
          (p A-p J) i * sourceSineCoefficient k F G A i) * Real.sin z := by
      simp only [dotProduct, sourceActiveMatrix_feature_apply,
        mul_add, Finset.sum_add_distrib, Finset.sum_mul]
      congr 1 <;> apply Finset.sum_congr rfl <;> intro i hi <;> ring
    _ = _ := by
      rw [hdotB, hdotU, hB A J,
        sourceSineCoefficient_anchor_projection K k p F G hF hG A J hAJ]

/-- The rational potential has exactly the source's quadratic cross
coefficient, in either anchor direction. -/
theorem sourceQuadraticPotential_metric_identity {m : ℕ} (k : ℕ)
    (hk : 0 < k) (A J : Fin m) (hAJ : A ≠ J) :
    4 * sourceMacroStep (k : ℝ) * sourceQuadraticPotential k A J *
      (sourceTheta k)^2 =
        ((9/8 : ℝ) * sourceDirectedScale k A J)^2 := by
  have hscale := source_theta_scale k hk
  change (sourceTheta k)^2 * (k : ℝ)^2 = 1/32 at hscale
  by_cases hlt : A.val < J.val
  · have hnot : ¬ J.val < A.val := by omega
    simp only [sourceQuadraticPotential, if_neg hnot, if_pos hlt,
      sourceDirectedScale]
    unfold sourceMacroStep
    nlinarith [hscale]
  · have hgt : J.val < A.val := by
      have hne : A.val ≠ J.val := fun h => hAJ (Fin.ext h)
      omega
    simp only [sourceQuadraticPotential, if_pos hgt, sourceDirectedScale,
      if_neg hlt]
    unfold sourceMacroStep
    nlinarith [hscale]

/-- Both literal directed projections have the source's exact linear and
quadratic digit terms. This is the algebraic input to the actual Euclidean
cross-metric identity. -/
theorem sourceActiveMatrix_projection_metric {n : ℕ} (K : ℝ) (k : ℕ)
    (hk : 0 < k)
    (p : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (B : Fin (n+1) → EuclideanSpace ℝ (Fin n))
    (F G : Fin n → EuclideanSpace ℝ (Fin n))
    (hB : ∀ A J, inner ℝ (p A-p J) (B A) =
      sourceQuadraticPotential k A J)
    (hF : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (F a) =
      fineIncidence n K i a)
    (hG : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (G a) =
      coarseIncidence n K i a)
    (A J : Fin (n+1)) (hAJ : A ≠ J) (z : ℝ) :
    let α := ((9/8 : ℝ) * sourceDirectedScale k A J) *
      Real.sin z / sourceTheta k
    2 * sourceMacroStep (k : ℝ) *
      ((fun i => (p A-p J) i) ⬝ᵥ
        (sourceActiveMatrix k B F G *ᵥ clusteredFeature A z)) =
      2 * (sourceMacroStep (k : ℝ) * colexDistance K A J) * α + α^2 := by
  rw [sourceActiveMatrix_anchor_projection K k p B F G hB hF hG A J hAJ z]
  dsimp
  rw [Real.cos_two_mul_eq_one_sub]
  have hθ : sourceTheta k ≠ 0 := (sourceTheta_pos k hk).ne'
  have hq := sourceQuadraticPotential_metric_identity k hk A J hAJ
  field_simp
  nlinarith [hq]

theorem sourceActiveMatrix_forward_projection_metric {n : ℕ} (K : ℝ) (k : ℕ)
    (hk : 0 < k)

      coarseIncidence n K i a)

      ((fun a => (p A-p J) a) ⬝ᵥ
        (sourceActiveMatrix k B F G *ᵥ clusteredFeature A

      2 * (sourceMacroStep (k : ℝ) * colexDistance K A J) *

  have h := sourceActiveMatrix_projection_metric K k hk p B F G
    hB hF hG A J (Fin.ne_of_lt hAJ)
    (sourceTheta k * ((i : ℝ)-((k : ℝ)-1)/2))
  simp [sourceAlpha, sourceDigit, sineDigit, sourceTheta,
    sourceDirectedScale, hAJ] at h ⊢
  convert h using 1; ring

/-- The off-diagonal flattening is canceled exactly by the private radius. -/
theorem source_macro_base_square {m k : ℕ} (hk : 0 < k)
    (K : ℝ) (p : Fin (m+1) → EuclideanSpace ℝ (Fin m))
    (hpd : ∀ i j,
      ‖p j-p i‖ ^ 2 =
        offDiagonalShift (colexSqData (m:=m) K)
          (sourceFlatteningRadius (m+1) k) i j)
    (A J : Fin (m+1)) (hAJ : A ≠ J) :
    (sourceMacroStep (k : ℝ)) ^ 2 *
      (∑ q : Fin m, ((p A-p J) q)^2) +
      4 * (sourceRadius (m+1) k)^2 =
      (sourceMacroStep (k : ℝ) * colexDistance K A J)^2 := by
  have hnorm := hpd J A
  rw [EuclideanSpace.real_norm_sq_eq] at hnorm
  have hne : J ≠ A := Ne.symm hAJ
  simp only [offDiagonalShift, if_neg hne, colexSqData,
    colexDistance_symm K J A] at hnorm
  have hS : sourceMacroStep (k : ℝ) ≠ 0 := by
    unfold sourceMacroStep
    have hk' : (k : ℝ) ≠ 0 := by exact_mod_cast (Nat.ne_of_gt hk)
    positivity
  have hr : (sourceMacroStep (k : ℝ))^2 *
      sourceFlatteningRadius (m+1) k =
      2 * (sourceRadius (m+1) k)^2 := by
    unfold sourceFlatteningRadius
    field_simp
  nlinarith [hnorm, hr]

theorem _root_.solution {m k : ℕ} (hk : 0 < k)
    (K : ℝ) (p : Fin (m+1) → EuclideanSpace ℝ (Fin m))
    (B : Fin (m+1) → EuclideanSpace ℝ (Fin m))
    (F G : Fin m → EuclideanSpace ℝ (Fin m))
    (V : Matrix (Fin (m+1) × Fin 5) (Fin (m+1) × Fin 5) ℝ)
    (hpd : ∀ i j,
      ‖p j-p i‖ ^ 2 =
        offDiagonalShift (colexSqData (m:=m) K)
          (sourceFlatteningRadius (m+1) k) i j)
    (hB : ∀ A J, inner ℝ (p A-p J) (B A) =
      sourceQuadraticPotential k A J)
    (hF : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (F a) =
      fineIncidence m K i a)
    (hG : ∀ i a, inner ℝ (p i.succ-p i.castSucc) (G a) =
      coarseIncidence m K i a)
    (b₀ b₁ : ℝ)
    (hfactor : (sourceActiveMatrix k B F G)ᵀ *
        (sourceActiveMatrix k B F G) + Vᵀ * V =
        Matrix.diagonal (fun q : Fin (m+1) × Fin 5 =>
          trigWeight b₀ b₁ b₀ q.2))
    (hreserve : 2*b₀^2+b₁^2 ≤ (sourceRadius (m+1) k)^2)
    (A J : Fin (m+1)) (hAJ : A.val < J.val)
    (i j : ℕ) (yi yj : ℝ) :
    let R := sourceRadius (m+1) k
    let axis := Real.sqrt (R^2-(2*b₀^2+b₁^2))
    let z (t : ℕ) := sourceTheta k * ((t : ℝ)-((k : ℝ)-1)/2)
    dist (clusterPoint (fun A q => p A q) (sourceActiveMatrix k B F G)
        V (sourceMacroStep (k : ℝ)) R axis A (z i) yi)
      (clusterPoint (fun A q => p A q) (sourceActiveMatrix k B F G)
        V (sourceMacroStep (k : ℝ)) R axis J (z j) yj)^2 =
      (sourceCrossBaseline K k (sourcePairCode A J) +
        sourceAlpha k i + sourceBeta k j)^2 -
        2*sourceAlpha k i*sourceBeta k j := by
  dsimp only
  have hne : A ≠ J := Fin.ne_of_lt hAJ
  have haxis :
      (Real.sqrt ((sourceRadius (m+1) k)^2-(2*b₀^2+b₁^2)))^2 =
      (sourceRadius (m+1) k)^2-(b₀^2+b₁^2+b₀^2) := by
    rw [Real.sq_sqrt (by linarith :
      0 ≤ (sourceRadius (m+1) k)^2-(2*b₀^2+b₁^2))]
    ring
  have hbase :
      (sourceMacroStep (k : ℝ))^2 *
        (∑ q : Fin m, (p A q-p J q)^2) +
        4*(sourceRadius (m+1) k)^2 =
      (sourceCrossBaseline K k (sourcePairCode A J))^2 := by
    rw [sourceCrossBaseline_pairCode K k A J hAJ]
    simpa only [WithLp.ofLp_sub, Pi.sub_apply] using
      source_macro_base_square hk K p hpd A J hne
  have hprojA := sourceActiveMatrix_forward_projection_metric K k hk
    p B F G hB hF hG A J hAJ i
  have hprojB := sourceActiveMatrix_reverse_projection_metric K k hk
    p B F G hB hF hG J A hAJ j
  have hprojA' :
      2 * sourceMacroStep (k : ℝ) *
        ((fun q => p A q-p J q) ⬝ᵥ
          (sourceActiveMatrix k B F G *ᵥ clusteredFeature A
            (sourceTheta k * ((i : ℝ)-((k : ℝ)-1)/2)))) =
      2 * sourceCrossBaseline K k (sourcePairCode A J) *
        sourceAlpha k i + (sourceAlpha k i)^2 := by
    rw [sourceCrossBaseline_pairCode K k A J hAJ]
    simpa only [WithLp.ofLp_sub, Pi.sub_apply, mul_assoc] using hprojA
  have hprojB' :
      -2 * sourceMacroStep (k : ℝ) *
        ((fun q => p A q-p J q) ⬝ᵥ
          (sourceActiveMatrix k B F G *ᵥ clusteredFeature J
            (sourceTheta k * ((j : ℝ)-((k : ℝ)-1)/2)))) =
      2 * sourceCrossBaseline K k (sourcePairCode A J) *
        sourceBeta k j + (sourceBeta k j)^2 := by
    rw [sourceCrossBaseline_pairCode K k A J hAJ,
      colexDistance_symm K A J]
    have hswap : ∀ q : Fin m, (p J-p A) q = -(p A-p J) q := by
      intro q
      simp only [WithLp.ofLp_sub, Pi.sub_apply]
      ring
    simp_rw [hswap] at hprojB
    simpa only [WithLp.ofLp_sub, Pi.sub_apply, dotProduct,
      neg_mul, Finset.sum_neg_distrib, mul_neg, neg_mul] using hprojB
  exact clusterPoint_cross_metric_square (fun A q => p A q)
    (sourceActiveMatrix k B F G) V
    (sourceMacroStep (k : ℝ)) (sourceRadius (m+1) k)
    (Real.sqrt ((sourceRadius (m+1) k)^2-(2*b₀^2+b₁^2)))
    b₀ b₁ b₀
    (sourceCrossBaseline K k (sourcePairCode A J))
    (sourceAlpha k i) (sourceBeta k j)
    hfactor A J hne _ _ yi yj haxis hbase hprojA' hprojB'
